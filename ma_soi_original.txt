@dataclass
class Person:
    id: int
    name: str
    role: str
    original: str = ""
    alive: bool = True
    revealed: bool = False
    no_vote: bool = False
    lover: int | None = None
    model: int | None = None
    enchanted: bool = False
    last_guard: int | None = None
    heal: bool = True
    poison: bool = True

    seen: dict = field(default_factory=dict)
    pack: set = field(default_factory=set)
    suspicion: dict = field(default_factory=dict)
    enchanted_known: set = field(default_factory=set)

    talks: int = 0
    style: str = ""

    @property
    def wolf(self):
        return self.role in ("W", "D")

# ENGINE

# Dùng generator/coroutine để chờ hành động mà không khóa GUI.

class Game:
    def __init__(self, app, roles, moderator, human_name, reveal):
        self.app = app
        self.rng = random.Random()
        self.gm = moderator
        self.human = None if moderator else 1
        self.reveal_dead = reveal

        self.rng.shuffle(roles)
        names = self.rng.sample(NAMES, len(roles))
        self.people = {}

        for i, role in enumerate(roles, 1):
            name = human_name if i == self.human else names[i - 1]
            p = Person(i, name, role, original=role)
            p.style = self.rng.choice(
                ["thận trọng", "thẳng thắn", "hài hước", "đa nghi"]
            )
            self.people[i] = p

        for p in self.people.values():
            p.suspicion = {
                i: self.rng.uniform(-0.5, 0.5)
                for i in self.people if i != p.id
            }

        self.day = 0
        self.phase = "Chuẩn bị"
        self.light = 0.0
        self.clock = 0.0
        self.deadline = None
        self.paused = False
        self.done = False
        self.wait = 0.0
        self.answer = None

        self.history = []
        self.claims = {}
        self.reports = []
        self.heard = set()
        self.last_votes = {}
        self.ballots = []
        self.lovers = None

        self.effects = []
        self.banner = ""
        self.banner_until = 0.0
        self.last_chat = -10.0

        self.flow = self.run()

    def p(self, i):
        return self.people[i]

    def name(self, i):
        return f"#{i} {self.p(i).name}"

    def live(self):
        return [i for i, p in self.people.items() if p.alive]

    def role_ids(self, role):
        return [i for i in self.live() if self.p(i).role == role]

    def options(self, ids):
        return [(i, self.name(i)) for i in ids]

    def visible(self, scope):
        return scope is None or self.gm or scope == self.human

    def log(self, text, scope=None, chat=False):
        item = {
            "time": round(self.clock, 1),
            "day": self.day,
            "phase": self.phase,
            "scope": scope,
            "chat": chat,
            "text": text,
        }
        self.history.append(item)
        if self.visible(scope):
            prefix = f"[{self.day:02d} · {self.phase}] "
            if scope is not None:
                prefix += "[KÍN] "
            self.app.append(prefix + text, chat)

    def private(self, i, text):
        self.log(text, i)

    def set_phase(self, text, day=False):
        self.phase = text
        self.light = float(day)
        self.deadline = None
        self.log("──── " + text + " ────")

    def tick(self, dt):
        if self.paused or self.done:
            return
        self.clock += dt
        self.wait -= dt
        if self.wait > 0:
            return
        try:
            self.wait = max(0.02, float(next(self.flow)))
        except StopIteration:
            self.done = True
            self.app.clear_action()

    # --------------------------------------------------------
    # NIỀM TIN RIÊNG VÀ HỘI THOẠI
    # --------------------------------------------------------

    def suspicion(self, actor, target):
        """Không tra vai ẩn của target để suy luận."""
        a = self.p(actor)
        if target == actor:
            return -100
        if target == a.lover:
            return -80
        if a.wolf and target in a.pack:
            return -15
        if target in a.seen:
            return 12 if a.seen[target] else -4

        t = self.p(target)
        if t.revealed:
            return 15 if t.wolf else -10

        return a.suspicion.get(target, 0)

    def suspect(self, actor, choices):
        if not choices:
            return None
        return max(
            choices,
            key=lambda i: self.suspicion(actor, i)
            + self.rng.uniform(-0.45, 0.45),
        )

    def hear(self, speaker, target=None, result=None,
             claim=False, accusation=False):
        if claim:
            self.claims[speaker] = "S"

        if target not in self.people:
            return

        if result is not None:
            evidence = (speaker, target, result)
            if evidence in self.reports:
                return

            contradict = any(
                s == speaker and t == target and r != result
                for s, t, r in self.reports
            )
            self.reports.append(evidence)

            for a in self.people.values():
                if not a.alive or a.id == speaker:
                    continue

                if contradict:
                    a.suspicion[speaker] = (
                        a.suspicion.get(speaker, 0) + 3
                    )

                if a.id != target:
                    distrust = max(0, self.suspicion(a.id, speaker))
                    trust = max(0.15, 1 - 0.12 * distrust)
                    change = (2.2 if result else -1.1) * trust
                    a.suspicion[target] = (
                        a.suspicion.get(target, 0) + change
                    )

        elif accusation:
            # Một người lặp cùng lời tố trong ngày không tăng điểm vô hạn.
            key = (self.day, speaker, target)
            if key in self.heard:
                return
            self.heard.add(key)

            for a in self.people.values():
                if a.alive and a.id not in (speaker, target):
                    distrust = max(0, self.suspicion(a.id, speaker))
                    change = max(0.04, 0.3 - distrust * 0.03)
                    a.suspicion[target] = (
                        a.suspicion.get(target, 0) + change
                    )

    def say(self, speaker, text, **evidence):
        self.p(speaker).talks += 1
        self.log(f"{self.name(speaker)}: {text}", chat=True)
        self.hear(speaker, **evidence)

    def bot_talk(self, actor):
        a = self.p(actor)
        others = [i for i in self.live() if i != actor]
        if not others:
            return
        target = self.suspect(actor, others)

        # Tiên tri dùng thông tin thật của chính mình.
        if a.role == "S":
            known = [i for i in others if i in a.seen and i != a.lover]
            bad = [i for i in known if a.seen[i]]
            chance = 0.78 if bad else 0.2
            if known and self.rng.random() < chance:
                t = self.rng.choice(bad or known)
                result = a.seen[t]
                text = (
                    f"Mình nhận Tiên tri. Đã soi {self.name(t)} "
                    + ("ra Sói." if result else "không phải Sói.")
                    + " Nhưng vẫn nên đối chiếu phiếu bầu."
                )
                self.say(actor, text, target=t, result=result, claim=True)
                return

        # Sói giả Tiên tri: chọn dựa trên niềm tin/hành vi công khai,
        # không nhìn vai ẩn của nạn nhân.
        if a.wolf:
            choices = [
                i for i in others if i not in a.pack and i != a.lover
            ]
            chance = 0.4 if actor in self.claims else 0.16
            if choices and self.rng.random() < chance:
                t = self.suspect(actor, choices)
                previous = [
                    r for s, x, r in self.reports if s == actor and x == t
                ]
                result = previous[-1] if previous else True
                text = (
                    f"Mình mới là Tiên tri. {self.name(t)} "
                    + ("ra Sói." if result else "ra không Sói.")
                    + " Nhận vai trước chưa chắc là thật đâu."
                )
                self.say(actor, text, target=t, result=result, claim=True)
                return

        if a.lover in others and self.rng.random() < 0.18:
            self.say(
                actor,
                f"Mình chưa muốn treo {self.name(a.lover)}. "
                "Bằng chứng như vậy còn mỏng, xem hướng khác thử."
            )
            return

        attackers = [
            s for s, t, result in self.reports
            if t == actor and result and self.p(s).alive
        ]
        if attackers and self.rng.random() < 0.45:
            t = attackers[-1]
            self.say(
                actor,
                f"{self.name(t)} nói soi mình ra Sói thì mình phản đối. "
                "Ai cũng có thể nhận Tiên tri; đừng tin chỉ vì họ nói chắc.",
                target=t, accusation=True,
            )
            return

        previous_vote = self.last_votes.get(target)
        if previous_vote:
            reason = (
                f"lượt trước bạn ấy bỏ phiếu cho {self.name(previous_vote)}"
            )
        elif target in self.claims:
            reason = "lời nhận Tiên tri đó chưa được xác thực"
        elif self.p(target).talks < 2:
            reason = "bạn ấy ít nói nên mình chưa có đủ dữ kiện"
        elif self.suspicion(actor, target) > 2:
            reason = "những lời tố cáo đang cùng hướng về bạn ấy"
        else:
            reason = "lập luận vừa rồi chưa đủ thuyết phục mình"

        templates = {
            "thận trọng": [
                f"Mình hơi nghi {self.name(target)} vì {reason}. "
                "Chưa chốt nhé, bạn phản biện thử.",
                f"Đừng chạy theo số đông. Với {self.name(target)}, "
                f"{reason}; mình muốn nghe thêm.",
            ],
            "thẳng thắn": [
                f"Hiện mình nghi nhất {self.name(target)}: {reason}. "
                "Không có thông tin mới thì mình vote hướng này.",
                f"{self.name(target)}, bạn đang nghi ai và vì sao? "
                "Nói rõ để mọi người còn đối chiếu.",
            ],
            "hài hước": [
                f"Làng mình họp căng quá rồi! Mình vẫn lăn tăn "
                f"{self.name(target)}, vì {reason}.",
                f"Đừng treo bằng cảm xúc nha. Mình đang nhìn "
                f"{self.name(target)}: {reason}.",
            ],
            "đa nghi": [
                f"Mình chưa tin lời nhận vai nào hết. "
                f"{self.name(target)} đáng xem lại vì {reason}.",
                f"Có thể Sói đang đẩy phiếu theo đám đông. "
                f"Mình tạm nghi {self.name(target)}; {reason}.",
            ],
        }

        self.say(
            actor, self.rng.choice(templates[a.style]),
            target=target, accusation=True,
        )

    def human_chat(self, text):
        if (
            self.human is None or self.done or self.paused
            or not self.p(self.human).alive
            or self.phase not in ("Thảo luận", "Bỏ phiếu", "Bỏ phiếu lại")
            or self.clock - self.last_chat < 0.8
        ):
            return False

        text = text.strip()[:600]
        if not text:
            return False

        self.last_chat = self.clock
        lower = text.lower()
        match = re.search(r"(?:#|số\s+)(\d+)", lower)
        target = int(match.group(1)) if match else None
        if target not in self.people:
            target = None

        claim = "tiên tri" in lower and any(
            s in lower for s in ("mình là", "tôi là", "mình nhận", "tôi nhận")
        )

        result = None
        if target is not None and "soi" in lower:
            if any(s in lower for s in ("không phải sói", "không sói", "ra dân")):
                result = False
            elif "sói" in lower:
                result = True

        accusation = target is not None and any(
            s in lower for s in ("nghi", "treo", "vote", "là sói")
        )

        self.say(
            self.human, text, target=target, result=result,
            claim=claim, accusation=accusation,
        )
        return True

    # --------------------------------------------------------
    # QUYẾT ĐỊNH KỸ NĂNG
    # --------------------------------------------------------

    def mixed_lovers(self):
        return bool(
            self.lovers
            and self.p(self.lovers[0]).wolf != self.p(self.lovers[1]).wolf
        )

    def bot_pick(self, actor, options, kind, maximum=1, meta=None):
        a = self.p(actor)
        ids = [key for key, _ in options]
        meta = meta or {}
        if not ids:
            return []

        if kind in ("cupid", "piper"):
            return self.rng.sample(ids, min(maximum, len(ids)))

        if kind == "model":
            return [self.rng.choice(ids)]

        if kind == "wolf":
            safe = [i for i in ids if i != a.lover] or ids
            # Chỉ dựa vào nhận vai và mức hoạt động công khai.
            return [max(
                safe,
                key=lambda i: (
                    4 * (self.claims.get(i) == "S")
                    + self.p(i).talks * 0.12 + self.rng.random()
                ),
            )]

        if kind == "seer":
            unknown = [i for i in ids if i not in a.seen]
            return [self.suspect(actor, unknown or ids)]

        if kind == "guard":
            return [max(
                ids,
                key=lambda i: (
                    4 * (self.claims.get(i) == "S")
                    + 2 * (i == actor)
                    + 4 * (i == a.lover)
                    + self.rng.random()
                ),
            )]

        if kind == "witch":
            victim = meta.get("victim")
            if "save" in ids and victim is not None:
                if (
                    victim in (actor, a.lover)
                    or self.suspicion(actor, victim) < 2
                    or self.rng.random() < 0.25
                ):
                    return ["save"]

            poison_targets = [i for i in ids if isinstance(i, int)]
            if poison_targets:
                target = self.suspect(actor, poison_targets)
                if (
                    target != a.lover
                    and self.suspicion(actor, target) > 3.2
                    and self.rng.random() < 0.8
                ):
                    return [target]
            return []

        if kind in ("vote", "hunter"):
            choices = [i for i in ids if i != a.lover] or ids

            # Sói trong đôi khác phe có thể phản đàn để cứu tình nhân.
            if a.wolf and a.lover is not None and self.mixed_lovers():
                return [max(
                    choices,
                    key=lambda i: (
                        4 if i in a.pack else self.suspicion(actor, i)
                    ) + self.rng.uniform(-0.5, 0.5),
                )]

            return [self.suspect(actor, choices)]

        return [self.rng.choice(ids)]

    def choose(self, actor, title, options, kind,
               minimum=1, maximum=1, optional=False, meta=None):
        if not options:
            return []

        if actor == self.human:
            self.answer = None
            self.app.show_action(title, options, minimum, maximum, optional)
            while self.answer is None:
                yield 0.1
            selected = self.answer
            self.answer = None
            self.app.clear_action()
        else:
            yield self.rng.uniform(0.45, 0.9)
            selected = self.bot_pick(actor, options, kind, maximum, meta)

        names = dict(options)
        decision = ", ".join(names[i] for i in selected) or "Không dùng"
        self.private(actor, f"{title} → {decision}")
        return selected

    # --------------------------------------------------------
    # THẮNG THUA
    # --------------------------------------------------------

    def team(self, actor):
        p = self.p(actor)
        if self.mixed_lovers() and actor in self.lovers:
            return "Tình nhân"
        if p.role == "P":
            return "Thổi sáo"
        return "Sói" if p.wolf else "Dân"

    def winner(self):
        live = self.live()
        if not live:
            return "Hòa", "Không còn ai sống."

        if (
            self.mixed_lovers()
            and len(live) == 2
            and set(live) == set(self.lovers)
        ):
            return "Tình nhân", "Đôi khác phe là hai người sống cuối cùng."

        for i in self.role_ids("P"):
            if all(self.p(j).enchanted for j in live if j != i):
                return "Thổi sáo", "Tất cả người sống khác đã bị mê hoặc."

        wolves = sum(self.p(i).wolf for i in live)
        if wolves == 0:
            return "Dân", "Không còn Ma sói."
        if wolves >= len(live) - wolves:
            return "Sói", "Số Sói bằng hoặc vượt số người còn lại."

        return None

    def finish(self, result):
        team, reason = result
        self.phase = "Kết thúc"
        self.deadline = None
        self.done = True
        self.app.clear_action()

        self.log(f"🏆 {team.upper()} THẮNG — {reason}" if team != "Hòa"
                 else f"HÒA — {reason}")

        if self.human is not None:
            self.log(
                "Bạn thắng ván này!"
                if self.team(self.human) == team
                else "Bạn không thắng ván này."
            )

        self.banner = f"KẾT QUẢ: {team.upper()}"
        self.banner_until = float("inf")
        self.log("CÔNG KHAI VAI CUỐI VÁN:")

        for p in self.people.values():
            extra = (
                f" (ban đầu {ROLE[p.original]})"
                if p.original != p.role else ""
            )
            self.log(
                f"{self.name(p.id)}: {ROLE[p.role]}{extra} — "
                f"{'sống' if p.alive else 'đã chết'}"
            )

    # --------------------------------------------------------
    # CÁI CHẾT VÀ DÂY CHUYỀN
    # --------------------------------------------------------

    def resolve_deaths(self, deaths):
        queue = []

        def mark(i, cause):
            p = self.p(i)
            if not p.alive:
                return

            p.alive = False
            p.revealed = p.revealed or self.reveal_dead
            queue.append((i, cause))
            self.effects.append((i, self.clock))

            self.banner = f"{self.name(i)} đã chết"
            self.banner_until = self.clock + 4
            detail = f" — {ROLE[p.role]}" if p.revealed else ""
            self.log(f"☠ {self.name(i)} đã chết{detail}.")
            self.log(f"Nguyên nhân: {self.name(i)} — {cause}.", "gm")

            # Vai đã lộ trở thành bằng chứng công khai.
            if p.revealed:
                for speaker, target, result in self.reports:
                    if target != i:
                        continue
                    actual_scan = p.wolf and p.role != "D"
                    delta = -1.3 if result == actual_scan else 3.5
                    for a in self.people.values():
                        if a.alive and a.id != speaker:
                            a.suspicion[speaker] = (
                                a.suspicion.get(speaker, 0) + delta
                            )

                for voter, target in self.last_votes.items():
                    if target != i:
                        continue
                    for a in self.people.values():
                        if a.alive and a.id != voter:
                            delta = -1.3 if p.wolf else 0.2
                            a.suspicion[voter] = (
                                a.suspicion.get(voter, 0) + delta
                            )

        # Đánh dấu đồng thời các nạn nhân ban đầu.
        for i, cause in deaths.items():
            mark(i, cause)

        cursor = 0
        while cursor < len(queue):
            i, cause = queue[cursor]
            cursor += 1
            p = self.p(i)
            yield 0.75

            if p.lover is not None and self.p(p.lover).alive:
                mark(p.lover, "Chết theo người yêu")

            if p.role == "H" and cause != "Chết theo người yêu":
                targets = self.live()
                if targets:
                    p.revealed = True
                    self.log(f"{self.name(i)} kích hoạt súng Thợ săn.")
                    chosen = yield from self.choose(
                        i, "Thợ săn: chọn người để bắn",
                        self.options(targets), "hunter",
                    )
                    if chosen:
                        mark(chosen[0], "Thợ săn bắn")

        # Chuyển phe sau khi toàn bộ chuỗi chết đã được xử lý.
        for i in self.role_ids("K"):
            p = self.p(i)
            if p.model is not None and not self.p(p.model).alive:
                p.role = "W"
                self.private(
                    i,
                    "Hình mẫu đã chết. Bạn trở thành Ma sói, "
                    "thức cùng đàn từ đêm tiếp theo.",
                )
                if p.lover is not None:
                    for x in (i, p.lover):
                        self.private(
                            x,
                            "Phe của đôi tình nhân đã thay đổi. "
                            + ("Hai bạn phải sống sót cuối cùng."
                               if self.mixed_lovers()
                               else "Hai bạn hiện thắng theo cùng phe."),
                        )

    # --------------------------------------------------------
    # BAN ĐÊM
    # --------------------------------------------------------

    def night(self):
        self.set_phase("Ban đêm")
        yield 1

        if self.day == 1:
            for i in self.role_ids("C"):
                self.log("Gọi Cupid.", "gm")
                choices = [
                    j for j in self.live() if self.p(j).role != "P"
                ]
                pair = yield from self.choose(
                    i, "Cupid: ghép đúng hai người",
                    self.options(choices), "cupid",
                    minimum=2, maximum=2,
                )
                if len(pair) == 2:
                    a, b = pair
                    self.p(a).lover = b
                    self.p(b).lover = a
                    self.lovers = (a, b)
                    for x, y in ((a, b), (b, a)):
                        self.private(
                            x,
                            f"Người yêu: {self.name(y)}. Một chết, một chết theo. "
                            + ("Khác phe: phải là hai người sống cuối cùng."
                               if self.mixed_lovers()
                               else "Cùng phe: thắng theo phe hiện tại."),
                        )

            for i in self.role_ids("K"):
                self.log("Gọi Đứa trẻ hoang dã.", "gm")
                selected = yield from self.choose(
                    i, "Đứa trẻ hoang dã: chọn hình mẫu",
                    self.options([j for j in self.live() if j != i]),
                    "model",
                )
                if selected:
                    self.p(i).model = selected[0]

        protected = set()
        for i in self.role_ids("G"):
            self.log("Gọi Bảo vệ.", "gm")
            p = self.p(i)
            selected = yield from self.choose(
                i, "Bảo vệ: chọn người bảo vệ",
                self.options([
                    j for j in self.live() if j != p.last_guard
                ]),
                "guard",
            )
            if selected:
                p.last_guard = selected[0]
                protected.add(selected[0])

        wolves = [i for i in self.live() if self.p(i).wolf]
        proposals = []

        if wolves:
            self.log("Gọi Ma sói.", "gm")
            for i in wolves:
                self.p(i).pack = set(wolves)
                self.private(
                    i, "Đàn Sói: " + ", ".join(self.name(j) for j in wolves)
                )

            for i in wolves:
                selected = yield from self.choose(
                    i, "Ma sói: đề xuất nạn nhân",
                    self.options([j for j in self.live() if j not in wolves]),
                    "wolf",
                )
                if selected:
                    target = selected[0]
                    proposals.append(target)
                    for j in wolves:
                        if j != i:
                            self.private(
                                j,
                                f"Đồng đội {self.name(i)} muốn cắn "
                                f"{self.name(target)}.",
                            )

        victim = None
        if proposals:
            counts = Counter(proposals)
            highest = max(counts.values())
            victim = self.rng.choice([
                i for i, n in counts.items() if n == highest
            ])
            for i in wolves:
                self.private(i, f"Đàn chốt cắn {self.name(victim)}.")

        if victim in protected:
            self.log("Bảo vệ chặn được cú cắn.", "gm")
            victim = None

        for i in self.role_ids("S"):
            self.log("Gọi Tiên tri.", "gm")
            selected = yield from self.choose(
                i, "Tiên tri: chọn người để soi",
                self.options([j for j in self.live() if j != i]),
                "seer",
            )
            if selected:
                target = selected[0]
                result = self.p(target).wolf and self.p(target).role != "D"
                self.p(i).seen[target] = result
                self.private(
                    i, f"Soi {self.name(target)}: "
                    + ("SÓI." if result else "KHÔNG PHẢI SÓI."),
                )

        poisoned = set()
        for i in self.role_ids("F"):
            self.log("Gọi Phù thủy.", "gm")
            p = self.p(i)
            options = []
            visible_victim = victim if p.heal else None

            if p.heal:
                self.private(
                    i, "Nạn nhân cần cứu: "
                    + (self.name(victim) if victim is not None else "không có"),
                )
                if victim is not None:
                    options.append(("save", f"CỨU {self.name(victim)}"))

            if p.poison:
                options += [
                    (j, f"ĐỘC {self.name(j)}")
                    for j in self.live() if j != i
                ]

            selected = yield from self.choose(
                i, "Phù thủy: dùng tối đa một bình",
                options, "witch", minimum=0, optional=True,
                meta={"victim": visible_victim},
            )
            if selected:
                if selected[0] == "save":
                    p.heal = False
                    victim = None
                else:
                    p.poison = False
                    poisoned.add(selected[0])

        for i in self.role_ids("P"):
            self.log("Gọi Người thổi sáo.", "gm")
            targets = [
                j for j in self.live()
                if j != i and not self.p(j).enchanted
            ]
            selected = yield from self.choose(
                i, "Người thổi sáo: chọn tối đa hai người",
                self.options(targets), "piper",
                minimum=0, maximum=min(2, len(targets)), optional=True,
            )

            for target in selected:
                self.p(target).enchanted = True

            enchanted = {
                j for j in self.live() if self.p(j).enchanted
            }
            self.p(i).enchanted_known = set(enchanted)

            for target in enchanted:
                self.p(target).enchanted_known = set(enchanted)
                self.private(
                    target, "Những người bị mê hoặc: "
                    + ", ".join(self.name(j) for j in sorted(enchanted)),
                )

        self.set_phase("Bình minh", day=True)
        yield 1

        deaths = {i: "Phù thủy đầu độc" for i in poisoned}
        if victim is not None:
            deaths.setdefault(victim, "Sói cắn")

        if deaths:
            yield from self.resolve_deaths(deaths)
        else:
            self.log("☀ Đêm qua không có ai chết.")
            yield 1

        result = self.winner()
        if result:
            self.finish(result)
            return

        for i in self.role_ids("B"):
            live = self.live()
            pos = live.index(i)
            neighbors = {live[pos - 1], live[(pos + 1) % len(live)]}
            growl = any(self.p(j).wolf for j in neighbors)
            self.log(
                "🐻 Gấu gầm: có Sói cạnh Người thuần gấu!"
                if growl else "🐻 Gấu không gầm."
            )

        yield 0.8

    def discussion(self):
        self.set_phase("Thảo luận", day=True)
        self.deadline = self.clock + DAY_TIME
        self.log(f"Bắt đầu {DAY_TIME} giây thảo luận.")
        next_talk = self.clock + 1
        recent = []

        while self.clock < self.deadline:
            if self.clock >= next_talk:
                bots = [i for i in self.live() if i != self.human]
                choices = [i for i in bots if i not in recent[-2:]] or bots
                if choices:
                    speaker = self.rng.choice(choices)
                    self.bot_talk(speaker)
                    recent.append(speaker)
                next_talk = self.clock + self.rng.uniform(2.0, 3.5)
            yield 0.12

        self.deadline = None
        self.log("Hết giờ thảo luận.")

    def collect_votes(self, candidates, runoff=False):
        self.set_phase("Bỏ phiếu lại" if runoff else "Bỏ phiếu", day=True)
        self.deadline = self.clock + VOTE_TIME

        voters = [i for i in self.live() if not self.p(i).no_vote]
        votes = {}
        scheduled = {
            i: self.clock + self.rng.uniform(1.5, 24)
            for i in voters if i != self.human
        }

        human_pending = self.human in voters
        self.answer = None

        if human_pending:
            choices = [i for i in candidates if i != self.human]
            self.app.show_action(
                f"Chọn người treo — tối đa {VOTE_TIME}s",
                self.options(choices), 0, 1, True,
            )

        def cast(voter, target):
            votes[voter] = target
            if target is None:
                self.log(f"{self.name(voter)} bỏ phiếu trắng.")
            else:
                self.log(
                    f"{self.name(voter)} bỏ phiếu cho {self.name(target)}."
                )

        while self.clock < self.deadline:
            if human_pending and self.answer is not None:
                cast(self.human, self.answer[0] if self.answer else None)
                human_pending = False
                self.answer = None
                self.app.clear_action()

            for voter in [
                i for i, when in scheduled.items() if self.clock >= when
            ]:
                choices = [i for i in candidates if i != voter]
                selected = self.bot_pick(
                    voter, self.options(choices), "vote"
                )
                cast(voter, selected[0] if selected else None)
                del scheduled[voter]

            if len(votes) == len(voters):
                break
            yield 0.1

        for voter in voters:
            if voter not in votes:
                cast(voter, None)

        self.app.clear_action()
        self.answer = None
        self.deadline = None
        self.last_votes = dict(votes)
        self.ballots.append({
            "day": self.day, "runoff": runoff, "votes": dict(votes)
        })

        counts = Counter(i for i in votes.values() if i is not None)
        self.log(
            "Tổng phiếu: " + ", ".join(
                f"{self.name(i)}: {n}" for i, n in counts.most_common()
            ) if counts else "Không có phiếu hợp lệ."
        )

        yield 0.8
        return counts

    def lynch(self, target):
        p = self.p(target)
        if p.role == "I" and not p.no_vote:
            p.revealed = True
            p.no_vote = True
            self.log(
                f"{self.name(target)} là Chàng ngốc! Không chết, "
                "nhưng mất phiếu và được miễn treo về sau."
            )
            yield 1
        else:
            self.log(f"Quyết định treo: {self.name(target)}.")
            yield from self.resolve_deaths({target: "Treo cổ"})

    def vote_phase(self):
        candidates = [
            i for i in self.live()
            if not (self.p(i).role == "I" and self.p(i).no_vote)
        ]
        counts = yield from self.collect_votes(candidates)
        if not counts:
            self.log("Hôm nay không treo ai.")
            return

        maximum = max(counts.values())
        tied = [i for i, n in counts.items() if n == maximum]
        if len(tied) == 1:
            yield from self.lynch(tied[0])
            return

        scapegoats = self.role_ids("X")
        if scapegoats:
            i = scapegoats[0]
            self.p(i).revealed = True
            self.log(f"Hòa phiếu! Vật tế thần {self.name(i)} chết thay.")
            yield from self.resolve_deaths({i: "Vật tế thần"})
            return

        self.log(
            "Hòa phiếu: " + ", ".join(self.name(i) for i in tied)
            + ". Biện hộ rồi bỏ phiếu lại."
        )
        self.set_phase("Thảo luận", day=True)

        for i in tied:
            if i != self.human:
                self.say(
                    i,
                    "Mình phản đối việc treo mình lúc này. "
                    "Hãy xem ai mở hướng tố và ai chỉ đang hùa theo.",
                )
                yield 1.2

        if self.human in tied:
            self.log("Bạn có 8 giây gửi lời biện hộ.")
            self.deadline = self.clock + 8
            while self.clock < self.deadline:
                yield 0.1
            self.deadline = None

        counts = yield from self.collect_votes(tied, runoff=True)
        if not counts:
            self.log("Không treo ai.")
            return

        maximum = max(counts.values())
        tied = [i for i, n in counts.items() if n == maximum]
        if len(tied) == 1:
            yield from self.lynch(tied[0])
        else:
            self.log("Vẫn hòa phiếu. Hôm nay không treo ai.")

    def run(self):
        self.log(
            f"Ván bắt đầu: {len(self.people)} người. "
            + ("Bạn làm quản trò." if self.gm else "Bạn nhập vai.")
        )

        for p in self.people.values():
            self.private(
                p.id,
                f"{self.name(p.id)} nhận vai {ROLE[p.role]}. {DESC[p.role]}"
            )

        yield 1
        while not self.done:
            self.day += 1
            yield from self.night()
            if self.done:
                return

            yield from self.discussion()
            yield from self.vote_phase()

            result = self.winner()
            if result:
                self.finish(result)
                return
            yield 1

    def update_total(self, *_):
        if not hasattr(self, "total_label"):
            return
        try:
            total = sum(int(v.get()) for v in self.role_vars.values())
            wolves = int(self.role_vars["W"].get()) + int(
                self.role_vars["D"].get()
            )
            self.total_label.configure(
                text=f"Tổng: {total} người • Sói ban đầu: {wolves}"
            )
        except (ValueError, tk.TclError):
            pass

    def apply(self, *_):
        custom = self.size_var.get() == "Tùy chỉnh"
        allowed = set(ROLE) if self.mode_var.get() == "Nâng cao" else BASIC

        if not custom:
            preset = PRESET[self.mode_var.get()][int(self.size_var.get())]
            for r, var in self.role_vars.items():
                var.set(str(preset.get(r, 0)))

        for r, spin in self.role_spins.items():
            if r not in allowed:
                self.role_vars[r].set("0")
            spin.configure(
                state="normal" if custom and r in allowed else "disabled"
            )
        self.update_total()

    def start(self):
        try:
            counts = {r: int(v.get()) for r, v in self.role_vars.items()}
        except ValueError:
            messagebox.showerror("Cấu hình", "Hãy nhập số nguyên.")
            return

        total = sum(counts.values())
        wolves = counts["W"] + counts["D"]

        if any(n < 0 for n in counts.values()):
            messagebox.showerror("Cấu hình", "Không được nhập số âm.")
            return
        if not 5 <= total <= 30:
            messagebox.showerror("Cấu hình", "Cần từ 5 đến 30 người.")
            return
        if any(n > 1 for r, n in counts.items() if r not in ("V", "W")):
            messagebox.showerror(
                "Cấu hình", "Mỗi vai đặc biệt chỉ có tối đa một bản sao."
            )
            return
        if wolves < 1 or wolves >= total - wolves:
            messagebox.showerror(
                "Cấu hình",
                "Cần ít nhất một Sói và số Sói phải ít hơn số người còn lại.",
            )
            return

        roles = [r for r, n in counts.items() for _ in range(n)]
        moderator = self.play_var.get() == "Quản trò"
        name = self.name_var.get().strip()[:18] or "Bạn"
        reveal = self.reveal_var.get()

        self.new_view()
        self.build_room()
        self.sky = 0
        self.game = Game(self, roles, moderator, name, reveal)

    def show_action(self, title, options, minimum, maximum, optional):
        self.action_options = options
        self.action_min = minimum
        self.action_max = maximum

        self.action_title.configure(
            text=title + (" · Ctrl/Shift để chọn nhiều" if maximum > 1 else "")
        )
        self.action_list.configure(state="normal")
        self.action_list.delete(0, "end")
        self.action_list.configure(
            selectmode="extended" if maximum > 1 else "browse"
        )

        for _, label in options:
            self.action_list.insert("end", label)

        self.confirm_button.configure(state="normal")
        self.skip_button.configure(state="normal" if optional else "disabled")
        self.action_error.configure(text="")

    def clear_action(self):
        self.action_options = []
        self.action_list.configure(state="normal")
        self.action_list.delete(0, "end")
        self.action_list.configure(state="disabled")
        self.action_title.configure(text="Đang theo dõi diễn biến...")
        self.confirm_button.configure(state="disabled")
        self.skip_button.configure(state="disabled")
        self.action_error.configure(text="")

    def confirm_action(self):
        g = self.game
        if not g or g.paused or g.done:
            return

        indexes = self.action_list.curselection()
        if not self.action_min <= len(indexes) <= self.action_max:
            self.action_error.configure(
                text=f"Chọn {self.action_min}–{self.action_max} mục."
            )
            return

        g.answer = [self.action_options[i][0] for i in indexes]
        self.confirm_button.configure(state="disabled")
        self.skip_button.configure(state="disabled")
        self.action_list.configure(state="disabled")
        self.action_error.configure(text="Đã gửi.")

    def skip_action(self):
        g = self.game
        if not g or g.paused or g.done:
            return
        g.answer = []
        self.confirm_button.configure(state="disabled")
        self.skip_button.configure(state="disabled")
        self.action_list.configure(state="disabled")
        self.action_error.configure(text="Đã bỏ qua.")

    def send_chat(self):
        g = self.game
        if g and g.human_chat(self.chat_var.get()):
            self.chat_var.set("")
            self.chat_hint.configure(text="Đã gửi.", foreground=MUTED)
        else:
            self.chat_hint.configure(
                text="Chỉ được chat khi còn sống, đang ở ban ngày và không tạm dừng.",
                foreground="#ffb5a9",
            )

    def pause(self):
        g = self.game
        if g and not g.done:
            g.paused = not g.paused
            self.pause_button.configure(
                text="▶ Tiếp tục" if g.paused else "⏸ Tạm dừng"
            )

    def role_visible(self, p):
        g = self.game
        return g.gm or g.done or p.id == g.human or p.revealed

    def extra_info(self, p):
        g = self.game
        full = g.gm or g.done or p.id == g.human
        info = []

        if p.no_vote:
            info.append("Mất phiếu")
        if full:
            if p.lover is not None:
                info.append(f"Yêu #{p.lover}")
            if p.model is not None:
                info.append(f"Hình mẫu #{p.model}")
            if p.enchanted:
                info.append("Bị mê hoặc")
            if p.role == "G" and p.last_guard is not None:
                info.append(f"Bảo vệ trước: #{p.last_guard}")
            if p.role == "F":
                info.append(
                    f"Cứu:{int(p.heal)} Độc:{int(p.poison)}"
                )
            if p.seen:
                info.append("Soi " + ", ".join(
                    f"#{i}:{'Sói' if result else 'không Sói'}"
                    for i, result in p.seen.items()
                ))
            if p.original != p.role:
                info.append(f"Gốc: {ROLE[p.original]}")

        return " · ".join(info)

    def refresh_info(self):
        g = self.game
        if g is None:
            return

        remaining = ""
        if g.deadline is not None:
            seconds = max(0, math.ceil(g.deadline - g.clock))
            remaining = f" • {seconds:02d}s"

        status = (
            f"Ngày {g.day} • {g.phase}{remaining} "
            f"• Còn {len(g.live())}/{len(g.people)}"
        )

        if g.paused:
            status += " • TẠM DỪNG"

        self.status.configure(text=status)

        if g.gm:
            wolves = sum(g.p(i).wolf for i in g.live())

            self.private_label.configure(
                text=(
                    f"QUẢN TRÒ — Sói còn sống: {wolves}. "
                    "Xem tab Nhân vật để theo dõi vai, thuốc, "
                    "tình nhân, hình mẫu.\n"
                    "Tab Lịch sử có đầy đủ hành động kín; "
                    "tab Ghi chú cho ghi chép tay."
                )
            )
        else:
            p = g.p(g.human)
            detail = self.extra_info(p)

            state = (
                "Còn sống"
                if p.alive
                else "Đã chết — chỉ theo dõi"
            )

            text = (
                f"BẠN: #{p.id} {p.name} • {ROLE[p.role]} "
                f"• {state}\n"
                f"{DESC[p.role]}"
            )

            if detail:
                text += "\n" + detail

            self.private_label.configure(text=text)

        for p in g.people.values():
            key = str(p.id)
            shown_role = (
                ROLE[p.role]
                if self.role_visible(p)
                else "?"
            )
            extra = self.extra_info(p)

            # Sói biết đồng đội sau khi nhận mặt ban đêm.
            if (
                not self.role_visible(p)
                and g.human is not None
                and p.id in g.p(g.human).pack
            ):
                shown_role = "Đồng đội Sói"

            values = (
                p.id,
                p.name,
                shown_role,
                "Sống" if p.alive else "Đã chết",
                extra,
            )

            if not p.alive:
                tags = ("dead",)
            elif p.id == g.human:
                tags = ("self",)
            else:
                tags = ()

            if self.roster.exists(key):
                self.roster.item(
                    key,
                    values=values,
                    tags=tags,
                )
            else:
                self.roster.insert(
                    "",
                    "end",
                    iid=key,
                    values=values,
                    tags=tags,
                )

    @staticmethod
    def blend(a, b, fraction):
        fraction = max(0, min(1, fraction))

        aa = [int(a[i:i + 2], 16) for i in (1, 3, 5)]
        bb = [int(b[i:i + 2], 16) for i in (1, 3, 5)]

        values = [
            round(x + (y - x) * fraction)
            for x, y in zip(aa, bb)
        ]

        return "#" + "".join(f"{v:02x}" for v in values)

    def draw(self):
        g = self.game
        c = self.canvas

        width = max(100, c.winfo_width())
        height = max(100, c.winfo_height())

        c.delete("all")

        top = self.blend("#090d24", "#76b5dc", self.sky)
        bottom = self.blend("#27324e", "#e9cba1", self.sky)

        # Nền gradient chuyển mượt giữa ngày và đêm.
        for band in range(32):
            y1 = height * band / 32
            y2 = height * (band + 1) / 32 + 1
            color = self.blend(top, bottom, band / 31)

            c.create_rectangle(
                0,
                y1,
                width,
                y2,
                fill=color,
                outline=color,
            )

        # Trăng chuyển thành mặt trời theo màu nền.
        celestial = self.blend("#e0e6ff", "#ffe59a", self.sky)
        cx, cy = width - 65, 55

        c.create_oval(
            cx - 24,
            cy - 24,
            cx + 24,
            cy + 24,
            fill=celestial,
            outline="",
        )

        # Sao mờ dần khi trời sáng.
        if self.sky < 0.8:
            star_color = self.blend(
                "#c6d4ff",
                top,
                self.sky / 0.8,
            )

            for k in range(23):
                x = (k * 137 + 31) % max(1, int(width))
                y = (k * 53 + 17) % max(1, int(height * 0.55))

                c.create_oval(
                    x,
                    y,
                    x + 2,
                    y + 2,
                    fill=star_color,
                    outline="",
                )

        # Đồi phía sau làng.
        hill = self.blend("#142637", "#628d79", self.sky)

        c.create_polygon(
            0,
            height * 0.88,
            width * 0.2,
            height * 0.69,
            width * 0.45,
            height * 0.9,
            width * 0.72,
            height * 0.66,
            width,
            height * 0.85,
            width,
            height,
            0,
            height,
            fill=hill,
            outline="",
        )

        count = len(g.people)
        columns = 4 if count <= 10 else 5
        rows = math.ceil(count / columns)

        padding = 12
        gap = 8
        area_top = 100

        usable_h = max(120, height - area_top - 14)
        card_w = (
            width - padding * 2 - gap * (columns - 1)
        ) / columns
        card_h = min(
            84,
            (usable_h - gap * (rows - 1)) / rows,
        )

        latest_effect = {}

        for pid, started in g.effects:
            if g.clock - started < 2.5:
                latest_effect[pid] = started

        title = (
            "LÀNG THỨC GIẤC"
            if self.sky > 0.5
            else "ĐÊM XUỐNG TRÊN LÀNG"
        )

        c.create_text(
            16,
            23,
            text=title,
            anchor="w",
            fill="#f4f6ff",
            font=("Segoe UI", 14, "bold"),
        )

        c.create_text(
            16,
            50,
            text=(
                "Bạn là quản trò"
                if g.gm
                else "Giữ kín vai • Quan sát • Suy luận"
            ),
            anchor="w",
            fill="#edf3ff",
            font=("Segoe UI", 10),
        )

        # Thẻ nhân vật.
        for index, p in enumerate(g.people.values()):
            row, col = divmod(index, columns)

            x1 = padding + col * (card_w + gap)
            y1 = area_top + row * (card_h + gap)
            x2 = x1 + card_w
            y2 = y1 + card_h

            fill = "#1c3049" if p.alive else "#333943"
            border = ACCENT if p.id == g.human else "#778aa2"

            if p.id in latest_effect:
                age = g.clock - latest_effect[p.id]
                flash = (math.sin(age * 17) + 1) / 2

                fill = self.blend(
                    "#5b202b",
                    "#c94c59",
                    flash,
                )

            c.create_rectangle(
                x1,
                y1,
                x2,
                y2,
                fill=fill,
                outline=border,
                width=2,
            )

            short_name = p.name[:12]

            c.create_text(
                (x1 + x2) / 2,
                y1 + card_h * 0.28,
                text=f"#{p.id} {short_name}",
                fill=FG,
                font=("Segoe UI", 10, "bold"),
            )

            role_text = (
                ROLE[p.role]
                if self.role_visible(p)
                else "Vai bí mật"
            )

            if (
                not self.role_visible(p)

    def export(self):
        g = self.game

        if not g:
            return

        # Dừng đồng hồ trong lúc chọn file.
        old_pause = g.paused
        g.paused = True

        try:
            path = filedialog.asksaveasfilename(
                title="Xuất nhật ký ván",
                defaultextension=".json",
                initialfile=(
                    f"ma_soi_{datetime.now():%Y%m%d_%H%M%S}.json"
                ),
                filetypes=[("JSON", "*.json")],
            )

            if not path:
                return

            # Chỉ xuất thông tin người chơi được phép xem.
            visible_history = [
                item
                for item in g.history
                if g.visible(item["scope"])
            ]

            roster = []

            for p in g.people.values():
                roster.append({
                    "id": p.id,
                    "name": p.name,
                    "alive": p.alive,
                    "role": (
                        ROLE[p.role]
                        if self.role_visible(p)
                        else "?"
                    ),
                    "information": self.extra_info(p),
                })

            data = {
                "exported_at": datetime.now().isoformat(),
                "mode": "moderator" if g.gm else "player",
                "day": g.day,
                "phase": g.phase,
                "finished": g.done,
                "notes": self.notes_box.get("1.0", "end-1c"),
                "players": roster,
                "ballots": g.ballots,
                "history": visible_history,
            }

            with open(path, "w", encoding="utf-8") as file:
                json.dump(
                    data,
                    file,
                    ensure_ascii=False,
                    indent=2,
                )

            messagebox.showinfo(
                "Xuất nhật ký",
                "Đã lưu nhật ký thành công.",
            )

        except OSError as error:
            messagebox.showerror(
                "Không lưu được",
                str(error),
            )

        finally:
            if self.game is g:
                g.paused = old_pause

            self.last_tick = time.monotonic()

    def leave(self):
        g = self.game

        if not g:
            self.menu()
            return

        old_pause = g.paused
        g.paused = True

        accepted = messagebox.askyesno(
            "Rời phòng",
            "Rời ván hiện tại?\n"
            "Ghi chú và lịch sử chưa xuất sẽ bị mất.",
        )

        self.last_tick = time.monotonic()

        if accepted:
            self.menu()
        else:
            g.paused = old_pause

    def close(self):
        g = self.game

        if g and not g.done:
            old_pause = g.paused
            g.paused = True

            accepted = messagebox.askyesno(
                "Thoát game",
                "Thoát game và bỏ ván hiện tại?",
            )

            self.last_tick = time.monotonic()

            if not accepted:
                g.paused = old_pause
                return

        self.root.destroy()

    # --------------------------------------------------------
    # VÒNG LẶP GUI
    # --------------------------------------------------------

    def tick(self):
        now = time.monotonic()
        dt = max(0, min(now - self.last_tick, 0.25))
        self.last_tick = now

        g = self.game

        if g is not None:
            try:
                g.tick(dt)

                if not g.paused:
                    fraction = min(1, dt * 1.4)
                    self.sky += (g.light - self.sky) * fraction

                self.draw()

                if now - self.last_refresh > 0.2:
                    self.refresh_info()
                    self.last_refresh = now

            except Exception as error:
                import traceback

                traceback.print_exc()

                # Dừng ván khi có lỗi để tránh tiếp tục
                # với trạng thái trò chơi không nhất quán.
                g.paused = True
                g.done = True
                self.clear_action()

                messagebox.showerror(
                    "Lỗi trong ván",
                    f"{type(error).__name__}: {error}\n\n"
                    "Ván đã dừng. Bạn có thể xuất nhật ký rồi rời phòng.\n"
                    "Xem cửa sổ terminal để biết chi tiết lỗi.",
                )

        self.root.after(50, self.tick)

def ai2_suspicion(self, actor, target):
    a = self.p(actor)
    t = self.p(target)

    if target == actor:
        return -100.0

    if target == a.lover:
        return -80.0

    # Sói chỉ biết đồng đội sau khi nhận mặt.
    if a.wolf and target in a.pack:
        return -18.0

    # Chỉ được dùng vai của mục tiêu nếu vai đã công khai.
    if t.revealed:
        return 18.0 if t.wolf else -12.0

    value = a.suspicion.get(target, 0.0)

    # Kết quả soi là thông tin riêng của chính người soi.
    if target in a.seen:
        if a.seen[target]:
            return 16.0

        # Không tuyệt đối tin kết quả âm tính:
        # Sói ngụy trang vẫn có thể qua mặt Tiên tri.
        value = min(value, 0.0) - 4.5

    return value


def ai2_hear(
    self,
    speaker,
    target=None,
    result=None,
    claim=False,
    accusation=False,
):
    # Ghi nhận nhận vai công khai.
    new_claim = claim and self.claims.get(speaker) != "S"

    if claim:
        self.claims[speaker] = "S"

    if new_claim:
        claimants = [
            i for i, role in self.claims.items()
            if role == "S" and self.p(i).alive
        ]

        for a in self.people.values():
            if not a.alive or a.id == speaker:
                continue

            # Bộ luật chỉ có tối đa một Tiên tri.
            # Tiên tri thật biết người khác nhận cùng vai là giả.
            if a.role == "S":
                a.suspicion[speaker] = max(
                    7.0,
                    a.suspicion.get(speaker, 0.0),
                )

            elif len(claimants) > 1:
                # Người thường không biết ai thật:
                # nghi ngờ nhẹ tất cả, không tự chọn đúng Tiên tri.
                for candidate in claimants:
                    if candidate != a.id:
                        a.suspicion[candidate] = (
                            a.suspicion.get(candidate, 0.0) + 0.3
                        )

    if target not in self.people:
        return

    if result is not None:
        evidence = (speaker, target, result)

        # Không cộng uy tín vô hạn khi lặp cùng kết quả soi.
        if evidence in self.reports:
            return

        previous = [
            r for s, t, r in self.reports
            if s == speaker and t == target
        ]
        contradict = any(r != result for r in previous)

        self.reports.append(evidence)

        for a in self.people.values():
            if not a.alive or a.id == speaker:
                continue

            if contradict:
                a.suspicion[speaker] = (
                    a.suspicion.get(speaker, 0.0) + 4.0
                )

            # Bị tố soi ra Sói: Dân biết lời tố đó sai.
            # Chỉ kiểm tra vai của CHÍNH MÌNH.
            if a.id == target:
                if result and not a.wolf:
                    a.suspicion[speaker] = max(
                        7.0,
                        a.suspicion.get(speaker, 0.0),
                    )
                continue

            # Tiên tri thật không bị lời soi giả dẫn dắt.
            if a.role == "S":
                if self.claims.get(speaker) == "S":
                    a.suspicion[speaker] = max(
                        7.0,
                        a.suspicion.get(speaker, 0.0),
                    )
                continue

            distrust = max(0.0, self.suspicion(a.id, speaker))
            trust = max(0.12, 1.0 - distrust * 0.13)

            # Một người đã có uy tín qua diễn biến công khai
            # được tin hơn người chưa được kiểm chứng.
            reputation = a.suspicion.get(speaker, 0.0)
            if reputation < -1.0:
                trust = min(1.3, trust + 0.2)

            competing = sum(
                role == "S" and self.p(i).alive
                for i, role in self.claims.items()
            )

            if competing > 1:
                trust *= 0.65

            delta = (2.8 if result else -1.4) * trust
            a.suspicion[target] = (
                a.suspicion.get(target, 0.0) + delta
            )

    elif accusation:
        key = (self.day, speaker, target)

        if key in self.heard:
            return

        self.heard.add(key)

        for a in self.people.values():
            if not a.alive or a.id in (speaker, target):
                continue

            distrust = max(0.0, self.suspicion(a.id, speaker))

            # Lời nghi ngờ suông có trọng lượng thấp hơn kết quả soi.
            change = max(0.02, 0.14 - distrust * 0.02)

            a.suspicion[target] = (
                a.suspicion.get(target, 0.0) + change
            )


def ai2_bot_talk(self, actor):
    a = self.p(actor)
    others = [i for i in self.live() if i != actor]

    if not others:
        return

    if not hasattr(self, "_ai2_announcements"):
        self._ai2_announcements = set()

    # --------------------------------------------------------
    # TIÊN TRI: công bố thông tin quan trọng kịp thời.
    # --------------------------------------------------------

    if a.role == "S":
        known = [
            i for i in others
            if i in a.seen and i != a.lover
        ]

        bad = [i for i in known if a.seen[i]]

        fake_claimants = [
            i for i in others
            if self.claims.get(i) == "S"
        ]

        # Luôn công bố nếu phát hiện Sói.
        # Có người giả vai thì phản bác ngay.
        # Từ ngày 2 hoặc bàn ít người: chủ động cung cấp thông tin.
        should_reveal = (
            bool(bad)
            or bool(fake_claimants)
            or self.day >= 2
            or len(self.live()) <= 6
        )

        announcement = (self.day, actor, "seer")

        if (
            known
            and should_reveal
            and announcement not in self._ai2_announcements
        ):
            self._ai2_announcements.add(announcement)

            target = self.rng.choice(bad or known)
            result = a.seen[target]

            text = (
                f"Mình là Tiên tri. Mình đã soi {self.name(target)} "
                + ("ra Sói." if result else "không phải Sói.")
            )

            if fake_claimants:
                text += (
                    f" {self.name(fake_claimants[0])} cũng nhận Tiên tri, "
                    "nhưng bộ này chỉ có một Tiên tri."
                )

            text += " Hãy ghi lại kết quả và đối chiếu với vai lộ sau này."

            self.say(
                actor,
                text,
                target=target,
                result=result,
                claim=True,
            )
            return

    # --------------------------------------------------------
    # PHẢN BÁC LỜI SOI SAI NHẮM VÀO BẢN THÂN.
    # --------------------------------------------------------

    attackers = [
        speaker
        for speaker, target, result in self.reports
        if (
            target == actor
            and result
            and self.p(speaker).alive
            and speaker != actor
        )
    ]

    defense_key = (self.day, actor, "defense")

    if attackers and defense_key not in self._ai2_announcements:
        self._ai2_announcements.add(defense_key)

        target = attackers[-1]

        self.say(
            actor,
            f"{self.name(target)} nói soi mình ra Sói. "
            "Mình phản đối kết quả đó. Đừng coi lời nhận vai là bằng chứng thật; "
            "hãy kiểm tra các kết quả trước và phiếu của người này.",
            target=target,
            accusation=True,
        )
        return

    # --------------------------------------------------------
    # SÓI: vẫn nói dối, nhưng tránh cả đàn cùng giả Tiên tri.
    # --------------------------------------------------------

    if a.wolf:
        ally_claimants = [
            i for i in a.pack
            if (
                i != actor
                and self.p(i).alive
                and self.claims.get(i) == "S"
            )
        ]

        candidates = [
            i for i in others
            if i not in a.pack and i != a.lover
        ]

        fake_key = (self.day, actor, "fake")

        if (
            candidates
            and not ally_claimants
            and fake_key not in self._ai2_announcements
            and self.rng.random() < 0.12
        ):
            self._ai2_announcements.add(fake_key)

            target = self.suspect(actor, candidates)

            previous = [
                result
                for speaker, t, result in self.reports
                if speaker == actor and t == target
            ]

            result = previous[-1] if previous else True

            self.say(
                actor,
                f"Mình nhận Tiên tri. {self.name(target)} "
                + ("ra Sói." if result else "không phải Sói.")
                + " Mọi người có thể kiểm chứng lời mình.",
                target=target,
                result=result,
                claim=True,
            )
            return

    # --------------------------------------------------------
    # PHÂN TÍCH KẾT QUẢ SOI VÀ PHIẾU CÔNG KHAI.
    # --------------------------------------------------------

    target = self.suspect(actor, others)
    confidence = self.suspicion(actor, target)

    reports_against = [
        speaker
        for speaker, t, result in self.reports
        if (
            t == target
            and result
            and self.p(speaker).alive
            and self.suspicion(actor, speaker) < 2.5
        )
    ]

    if target in a.seen and a.seen[target]:
        reason = "mình có kết quả soi riêng xác nhận mục tiêu này"

    elif reports_against:
        reason = (
            f"{self.name(reports_against[-1])} đã công bố kết quả soi ra Sói, "
            "và mình chưa thấy bằng chứng đủ mạnh bác bỏ"
        )

    elif a.role == "S" and self.claims.get(target) == "S":
        reason = "người này nhận trùng vai Tiên tri của mình"

    elif self.last_votes.get(target) is not None:
        previous = self.last_votes[target]
        dead = self.p(previous)

        if dead.revealed and not dead.alive:
            reason = (
                f"lượt trước người này vote {self.name(previous)}, "
                f"sau đó mục tiêu lộ vai {ROLE[dead.role]}"
            )
        else:
            reason = (
                f"mình muốn đối chiếu việc người này vote "
                f"{self.name(previous)} ở lượt trước"
            )

    elif self.claims.get(target) == "S":
        reason = "lời nhận Tiên tri này chưa được kiểm chứng"

    else:
        reason = "mình chưa có bằng chứng mạnh, cần nghe giải thích thêm"

    if confidence > 3:
        openings = [
            "Mình nghi khá rõ",
            "Nếu chưa có phản chứng, mình ưu tiên kiểm tra",
            "Hướng vote hiện tại của mình là",
        ]
    else:
        openings = [
            "Mình mới tạm nghi",
            "Chưa chốt nhé, mình muốn hỏi",
            "Thông tin còn ít, mình đang xem xét",
        ]

    text = (
        f"{self.rng.choice(openings)} {self.name(target)}: {reason}. "
    )

    if a.style == "thận trọng":
        text += "Ai có thông tin ngược lại thì nói trước khi vote."
    elif a.style == "thẳng thắn":
        text += "Bạn phản biện trực tiếp điểm này nhé."
    elif a.style == "hài hước":
        text += "Treo nhầm là làng lại mất ngủ đấy, kiểm tra kỹ nha."
    else:
        text += "Mình cũng sẽ xem ai đang đẩy phiếu theo hướng này."

    self.say(
        actor,
        text,
        target=target,
        accusation=confidence > 0.5,
    )


def ai2_bot_pick(
    self,
    actor,
    options,
    kind,
    maximum=1,
    meta=None,
):
    a = self.p(actor)
    ids = [key for key, _ in options]
    meta = meta or {}

    if not ids:
        return []

    if kind in ("cupid", "piper"):
        return self.rng.sample(ids, min(maximum, len(ids)))

    if kind == "model":
        return [self.rng.choice(ids)]

    if kind == "seer":
        fresh = [i for i in ids if i not in a.seen]
        choices = fresh or ids

        # Ưu tiên kiểm chứng người đang bị tố và người nhận trùng vai.
        def investigation_value(i):
            value = self.suspicion(actor, i)

            if self.claims.get(i) == "S":
                value += 3.0

            value += self.rng.uniform(-0.3, 0.3)
            return value

        return [max(choices, key=investigation_value)]

    if kind == "guard":
        def protection_value(i):
            value = self.rng.uniform(0.0, 0.7)

            if i == a.lover:
                value += 8.0

            if i == actor:
                value += 1.2

            if self.claims.get(i) == "S":
                # Bảo vệ người nhận Tiên tri đáng tin hơn,
                # không tự biết ai là Tiên tri thật.
                distrust = max(0.0, self.suspicion(actor, i))
                value += max(0.0, 6.0 - distrust)

            if self.suspicion(actor, i) < -1.5:
                value += 1.0

            return value

        return [max(ids, key=protection_value)]

    if kind == "witch":
        victim = meta.get("victim")

        if "save" in ids and victim is not None:
            # Ưu tiên cứu mình hoặc người yêu.
            if victim in (actor, a.lover):
                return ["save"]

            # Thường cứu nạn nhân đầu; không giữ bình đến lúc quá muộn.
            # Nhưng nếu rất nghi nạn nhân thì có thể giữ lại.
            if self.suspicion(actor, victim) < 4.5:
                return ["save"]

        poison_targets = [
            i for i in ids
            if isinstance(i, int) and i != a.lover
        ]

        if poison_targets:
            target = self.suspect(actor, poison_targets)
            suspicion = self.suspicion(actor, target)

            credible_reports = [
                speaker
                for speaker, t, result in self.reports
                if (
                    t == target
                    and result
                    and self.p(speaker).alive
                    and self.suspicion(actor, speaker) < 1.0
                )
            ]

            # Không đầu độc chỉ vì vài câu tố cáo suông.
            if (
                suspicion >= 6.0
                or (credible_reports and suspicion >= 2.5)
            ):
                return [target]

        return []

    if kind == "wolf":
        candidates = [i for i in ids if i != a.lover] or ids

        def kill_value(i):
            value = self.p(i).talks * 0.08

            if self.claims.get(i) == "S":
                value += 3.0

            # Có dao động chiến thuật, không tối ưu tuyệt đối.
            value += self.rng.uniform(0.0, 2.0)
            return value

        return [max(candidates, key=kill_value)]

    if kind in ("vote", "hunter"):
        candidates = [i for i in ids if i != a.lover] or ids

        if a.wolf and a.lover is not None and self.mixed_lovers():
            return [max(
                candidates,
                key=lambda i: (
                    4.0 if i in a.pack else self.suspicion(actor, i)
                ) + self.rng.uniform(-0.3, 0.3),
            )]

        def vote_value(i):
            value = self.suspicion(actor, i)

            # Nếu thông tin rõ ràng, giảm yếu tố ngẫu nhiên.
            uncertainty = 0.15 if abs(value) >= 2.0 else 0.45
            return value + self.rng.uniform(-uncertainty, uncertainty)

        return [max(candidates, key=vote_value)]

    return [self.rng.choice(ids)]


def ai2_discussion(self):
    self.set_phase("Thảo luận", day=True)
    self.deadline = self.clock + DAY_TIME

    self.log(
        f"Bắt đầu {DAY_TIME} giây thảo luận. "
        "Mỗi bot được một lượt nói trước khi bắt đầu lượt tiếp theo."
    )

    bots = [i for i in self.live() if i != self.human]
    queue = bots[:]
    self.rng.shuffle(queue)

    next_talk = self.clock + 0.8
    previous_speaker = None

    # Bàn đông vẫn đủ thời gian cho mọi bot có ít nhất một lượt.
    interval = max(
        1.0,
        min(2.5, (DAY_TIME - 5) / max(1, len(bots))),
    )

    while self.clock < self.deadline:
        if self.clock >= next_talk and bots:
            if not queue:
                queue = bots[:]
                self.rng.shuffle(queue)

                if (
                    len(queue) > 1
                    and queue[0] == previous_speaker
                ):
                    queue[0], queue[1] = queue[1], queue[0]

            speaker = queue.pop(0)
            self.bot_talk(speaker)

            previous_speaker = speaker
            next_talk = self.clock + interval

        yield 0.1

    self.deadline = None
    self.log("Hết giờ thảo luận.")

import re as _chat_re
from collections import deque as _chat_deque


# Giữ lại hành vi AI trước khi vá.
_CHAT_OLD_SAY = Game.say
_CHAT_OLD_TALK = Game.bot_talk
_CHAT_OLD_HUMAN = Game.human_chat


def chat3_init(g):
    if hasattr(g, "_chat3_events"):
        return

    g._chat3_events = []
    g._chat3_pending = _chat_deque()
    g._chat3_replied = set()
    g._chat3_used = set()
    g._chat3_serial = 0

    # Mỗi người có giọng riêng, không phụ thuộc vai bí mật.
    voices = ["gắt", "cà khịa", "lạnh", "bộc trực", "đa nghi"]
    g._chat3_voice = {
        i: g.rng.choice(voices)
        for i in g.people
    }


def chat3_say(self, speaker, text, **evidence):
    chat3_init(self)

    # Giữ nguyên cơ chế nghe và cập nhật niềm tin của AI V2.
    _CHAT_OLD_SAY(self, speaker, text, **evidence)

    self._chat3_serial += 1

    target = evidence.get("target")
    if target is None:
        match = _chat_re.search(
            r"(?:#|số\s+)(\d+)", text.lower()
        )
        if match:
            candidate = int(match.group(1))
            if candidate in self.people:
                target = candidate

    event = {
        "id": self._chat3_serial,
        "day": self.day,
        "speaker": speaker,
        "target": target,
        "text": text,
        "result": evidence.get("result"),
        "claim": bool(evidence.get("claim")),
        "accusation": bool(evidence.get("accusation")),
        "question": "?" in text or any(
            word in text.lower()
            for word in (
                "tại sao", "vì sao", "giải thích",
                "nghĩ sao", "nghi ai", "nói rõ",
            )
        ),
    }

    self._chat3_events.append(event)

    # Chỉ cần giữ hội thoại gần đây để phản hồi.
    if len(self._chat3_events) > 240:
        self._chat3_events = self._chat3_events[-180:]


def chat3_human_chat(self, text):
    chat3_init(self)

    accepted = _CHAT_OLD_HUMAN(self, text)
    if not accepted:
        return False

    if not self._chat3_events:
        return True

    event = self._chat3_events[-1]

    # Ưu tiên phản hồi trong pha thảo luận.
    # Pha bỏ phiếu vẫn giữ đúng thời gian và cơ chế vote cũ.
    if self.phase != "Thảo luận":
        return True

    bots = [
        i for i in self.live()
        if i != self.human
    ]

    if not bots:
        return True

    responders = []
    target = event["target"]

    # Được người thật gọi tên thì trả lời trước.
    if target in bots:
        responders.append(target)

    others = [i for i in bots if i not in responders]
    self.rng.shuffle(others)

    # Một người trả lời trực tiếp và một người xen vào.
    responders.extend(others[:max(0, 2 - len(responders))])

    # Giới hạn hàng chờ; không để spam chat khóa toàn bộ lượt nói.
    while len(self._chat3_pending) > 4:
        self._chat3_pending.popleft()

    for actor in responders:
        self._chat3_pending.append((actor, event["id"]))

    return True


def chat3_unique(self, actor, sentences):
    """Tránh lặp nguyên câu đã dùng trong cùng ván."""
    chat3_init(self)

    options = list(sentences)
    self.rng.shuffle(options)

    for text in options:
        if text not in self._chat3_used:
            self._chat3_used.add(text)
            return text

    # Khi hết câu gốc, kết hợp giọng cá nhân và ngữ cảnh hiện tại.
    voice = self._chat3_voice[actor]

    prefixes = {
        "gắt": [
            "Khoan đã. ", "Nói cho rõ nhé. ", "Ê, từ từ. ",
            "Đừng đánh trống lảng. ",
        ],
        "cà khịa": [
            "Ủa rồi sao nữa? ", "Nghe căng đấy, nhưng ",
            "Vãi, làng này loạn thật. ", "Nói thế ai chả nói được. ",
        ],
        "lạnh": [
            "Nhìn lại dữ kiện đi. ", "Tách hai chuyện ra. ",
            "Mình chưa đồng ý. ", "Chốt một điểm thôi. ",
        ],
        "bộc trực": [
            "Nói thẳng nhé. ", "Không vòng vo nữa. ",
            "Mình hỏi thật. ", "Nghe đây đã. ",
        ],
        "đa nghi": [
            "Có gì đó không ổn. ", "Đừng hùa vội. ",
            "Mình thấy đáng ngờ đấy. ", "Ai đang hưởng lợi ở đây? ",
        ],
    }

    endings = [
        " Ai có ý khác thì phản biện đi.",
        " Đừng đổi chủ đề lúc này.",
        " Mình đang nghe câu trả lời.",
        " Chưa đủ để mình chốt phiếu.",
        " Có thông tin mới thì mình đổi ý.",
        " Đừng biến suy đoán thành sự thật.",
    ]

    for _ in range(60):
        text = (
            self.rng.choice(prefixes[voice])
            + self.rng.choice(options)
            + self.rng.choice(endings)
        )
        if text not in self._chat3_used:
            self._chat3_used.add(text)
            return text

    # Trường hợp rất hiếm khi đã dùng hết tổ hợp.
    return (
        f"Ngày {self.day}, mình vẫn giữ câu hỏi này: "
        + self.rng.choice(options)
    )


def chat3_find_event(self, actor, event_id=None):
    chat3_init(self)

    if event_id is not None:
        for event in reversed(self._chat3_events):
            if event["id"] == event_id:
                return event
        return None

    candidates = [
        e for e in self._chat3_events[-14:]
        if (
            e["day"] == self.day
            and e["speaker"] != actor
            and self.p(e["speaker"]).alive
            and (actor, e["id"]) not in self._chat3_replied
        )
    ]

    if not candidates:
        return None

    # Ưu tiên người vừa gọi tên mình.
    addressed = [
        e for e in candidates
        if e["target"] == actor
    ]

    if addressed:
        return addressed[-1]

    # Sau đó ưu tiên lời của người chơi thật.
    human_events = [
        e for e in candidates
        if e["speaker"] == self.human
    ]

    if human_events:
        return human_events[-1]

    return candidates[-1]

def chat3_bot_talk(self, actor, event_id=None):
    chat3_init(self)

    if not self.p(actor).alive:
        return

    a = self.p(actor)

    # Phản hồi trực tiếp người thật được ưu tiên.
    if event_id is not None:
        event = chat3_find_event(self, actor, event_id)
        if event and chat3_reply(self, actor, event):
            return

    # Không để phần tán gẫu che mất thông tin sống còn của Tiên tri.
    known_wolf = any(
        result
        and self.p(i).alive
        and i != a.lover
        for i, result in a.seen.items()
    )

    announcement_missing = (
        (self.day, actor, "seer")
        not in getattr(self, "_ai2_announcements", set())
    )

    if a.role == "S" and known_wolf and announcement_missing:
        _CHAT_OLD_TALK(self, actor)
        return

    event = chat3_find_event(self, actor)

    # Đa số lượt nói phải liên quan hội thoại đang diễn ra.
    if event is not None and self.rng.random() < 0.72:
        if chat3_reply(self, actor, event):
            return

    # Giữ một phần lượt nói cho suy luận, nhận vai và nói dối của V2.
    _CHAT_OLD_TALK(self, actor)


def chat3_discussion(self):
    chat3_init(self)

    self.set_phase("Thảo luận", day=True)
    self.deadline = self.clock + DAY_TIME

    # Không mang yêu cầu phản hồi từ ngày trước sang ngày mới.
    self._chat3_pending.clear()

    self.log(
        f"Bắt đầu {DAY_TIME} giây tranh luận. "
        "Gọi #số để bot đó ưu tiên trả lời. "
        "Chat mới không kéo dài đồng hồ."
    )

    bots = [i for i in self.live() if i != self.human]
    queue = bots[:]
    self.rng.shuffle(queue)

    interval = max(
        1.0,
        min(2.3, (DAY_TIME - 5) / max(1, len(bots))),
    )

    next_talk = self.clock + 0.8
    previous = None
    priority_streak = 0

    while self.clock < self.deadline:
        if self.clock >= next_talk and bots:
            chosen = None
            event_id = None

            # Tối đa hai lượt ưu tiên liên tiếp,
            # tránh chat người thật làm các bot khác mất hết lượt.
            if priority_streak < 2:
                while self._chat3_pending:
                    actor, eid = self._chat3_pending.popleft()

                    if (
                        self.p(actor).alive
                        and (actor, eid) not in self._chat3_replied
                    ):
                        chosen = actor
                        event_id = eid
                        priority_streak += 1
                        break

            if chosen is None:
                priority_streak = 0

                if not queue:
                    queue = bots[:]
                    self.rng.shuffle(queue)

                    if len(queue) > 1 and queue[0] == previous:
                        queue[0], queue[1] = queue[1], queue[0]

                chosen = queue.pop(0)

            self.bot_talk(chosen, event_id=event_id)
            previous = chosen
            next_talk = self.clock + interval

        yield 0.1

    self.deadline = None
    self._chat3_pending.clear()
    self.log("Hết giờ tranh luận. Chuyển sang bỏ phiếu.")

def v4_defense_talk(self, actor, event_id=None):
    """Chỉ được gọi cho bot thuộc nhóm đang hòa phiếu."""
    chat3_init(self)

    allowed = getattr(self, "_v4_defendants", set())

    if actor not in allowed or not self.p(actor).alive:
        return

    # Phản hồi lời vừa nói của ứng viên khác.
    events = [
        event
        for event in self._chat3_events
        if (
            event["id"] > self._v4_defense_start
            and event["speaker"] in allowed
            and event["speaker"] != actor
            and (actor, event["id"]) not in self._chat3_replied
        )
    ]

    chosen_event = None

    if event_id is not None:
        chosen_event = next(
            (event for event in events if event["id"] == event_id),
            None,
        )

    if chosen_event is None and events:
        addressed = [
            event for event in events
            if event["target"] == actor
        ]

        if addressed:
            chosen_event = addressed[-1]
        elif self.rng.random() < 0.7:
            chosen_event = events[-1]

    if chosen_event is not None:
        if chat3_reply(self, actor, chosen_event):
            return

    # Nếu chưa có câu để đáp, chủ động đưa lý do biện hộ.
    a = self.p(actor)

    others = [
        i for i in sorted(allowed)
        if i != actor and self.p(i).alive
    ]

    if not others:
        return

    target = self.suspect(actor, others)
    target_name = self.name(target)
    suspicion = self.suspicion(actor, target)

    # Tiên tri thật vẫn được công bố thông tin riêng.
    announcement_key = (self.day, actor, "runoff_seer")

    if not hasattr(self, "_v4_announced"):
        self._v4_announced = set()

    known = [
        i for i in self.live()
        if i != actor and i in a.seen and i != a.lover
    ]

    if (
        a.role == "S"
        and known
        and announcement_key not in self._v4_announced
    ):
        self._v4_announced.add(announcement_key)

        bad = [i for i in known if a.seen[i]]
        scanned = self.rng.choice(bad or known)
        result = a.seen[scanned]

        self.say(
            actor,
            f"Khoan chốt phiếu. Mình là Tiên tri, "
            f"đã soi {self.name(scanned)} "
            + ("ra Sói. " if result else "không phải Sói. ")
            + "Ai đang định treo mình thì xem lại thông tin này.",
            target=scanned,
            result=result,
            claim=True,
        )
        return

    old_vote = self.last_votes.get(actor)

    if old_vote is not None:
        vote_detail = (
            f"Lượt vừa rồi mình vote {self.name(old_vote)}. "
            "Mình chịu trách nhiệm cho hướng đó, "
            "không có chuyện phủi tay khi bị hỏi."
        )
    else:
        vote_detail = (
            "Lượt vừa rồi mình bỏ trắng vì chưa đủ chắc. "
            "Thiếu thông tin không có nghĩa mình là Sói."
        )

    if target == a.lover:
        lines = [
            "Mình không định cứu bản thân bằng cách tố bừa "
            f"{target_name}. Cả làng xem lại lý do đưa mình lên đây.",

            "Bị hòa phiếu không biến một người thành Sói. "
            "Ai chuẩn bị đổi phiếu thì nhớ đối chiếu lời tố ban đầu.",
        ]
        accuse = False

    elif suspicion > 2:
        lines = [
            f"{target_name}, mình vẫn nghi bạn. "
            "Đừng chỉ nói 'mình là Dân'; trả lời điểm bị chất vấn đi.",

            f"Giờ còn mấy người trên giàn thôi. {target_name}, "
            "nói rõ vì sao nên tin bạn hơn mình, đừng vòng vo nữa.",

            f"Mình không xin phiếu thương hại. "
            f"Hướng của mình vẫn là {target_name}. "
            "Ai nghe được phản chứng thì đưa ra trong đầu trước khi vote.",
        ]
        accuse = True

    else:
        lines = [
            f"{target_name}, bạn đang nghi ai trong nhóm này? "
            "Nói một hướng rõ ràng đi, đừng chỉ xin cả làng tha.",

            f"Mình chưa có bằng chứng chắc với {target_name}. "
            "Nhưng cũng không chấp nhận bị treo chỉ vì người khác nói to hơn.",

            f"Ê {target_name}, hai đứa cùng đang bị nghi. "
            "Mình muốn nghe lý do cụ thể, không phải thi xem ai nhận Dân hay hơn.",

            "Vãi, cứ hùa một lượt là mình lên giàn luôn. "
            "Người đang nghe nhớ phân biệt lời tố với bằng chứng.",

            vote_detail,
        ]
        accuse = False

    text = chat3_unique(self, actor, lines)

    self.say(
        actor,
        text,
        target=target,
        accusation=accuse,
    )


def v4_defense(self, tied):
    chat3_init(self)

    # Giữ tên pha Thảo luận để tương thích bộ xử lý chat cũ.
    # Banner và lịch sử sẽ chỉ rõ đây là biện hộ hòa phiếu.
    self.set_phase("Thảo luận", day=True)

    self._v4_defendants = set(tied)
    self._v4_defense_start = self._chat3_serial
    self._chat3_pending.clear()

    self.deadline = self.clock + DEFENSE_TIME
    self.banner = f"BIỆN HỘ HÒA PHIẾU • {DEFENSE_TIME} GIÂY"
    self.banner_until = self.deadline

    self.log(
        f"BIỆN HỘ {DEFENSE_TIME} GIÂY — chỉ "
        + ", ".join(self.name(i) for i in tied)
        + " được nói. Những người khác lắng nghe và chuẩn bị vote lại."
    )

    bots = [
        i for i in tied
        if i != self.human and self.p(i).alive
    ]

    queue = bots[:]
    self.rng.shuffle(queue)

    # Bàn nhiều ứng viên vẫn cố gắng cho mỗi bot một lượt.
    interval = max(
        0.65,
        min(2.1, (DEFENSE_TIME - 2) / max(1, len(bots))),
    )

    next_talk = self.clock + 0.6
    priority_streak = 0

    try:
        while self.clock < self.deadline:
            if self.clock >= next_talk and bots:
                actor = None
                event_id = None

                # Chỉ xử lý yêu cầu phản hồi thuộc nhóm hòa.
                # Xen tối đa một lượt ưu tiên giữa các lượt thường.
                if priority_streak < 1:
                    while self._chat3_pending:
                        candidate, eid = self._chat3_pending.popleft()

                        if (
                            candidate in bots
                            and (candidate, eid)
                            not in self._chat3_replied
                        ):
                            actor = candidate
                            event_id = eid
                            priority_streak += 1
                            break

                if actor is None:
                    priority_streak = 0

                    if not queue:
                        queue = bots[:]
                        self.rng.shuffle(queue)

                    actor = queue.pop(0)

                v4_defense_talk(self, actor, event_id)
                next_talk = self.clock + interval

            yield 0.1

    finally:
        self._v4_defendants = None
        self._chat3_pending.clear()
        self.deadline = None
        self.banner_until = self.clock

    self.log(
        f"Hết {DEFENSE_TIME} giây biện hộ. Bắt đầu bỏ phiếu lại."
    )


def v4_vote_phase(self):
    candidates = [
        i for i in self.live()
        if not (
            self.p(i).role == "I"
            and self.p(i).no_vote
        )
    ]

    counts = yield from self.collect_votes(candidates)

    if not counts:
        self.log("Hôm nay không treo ai.")
        return

    highest = max(counts.values())
    tied = [
        i for i, count in counts.items()
        if count == highest
    ]

    if len(tied) == 1:
        yield from self.lynch(tied[0])
        return

    # Giữ luật Vật tế thần hiện tại.
    scapegoats = self.role_ids("X")

    if scapegoats:
        target = scapegoats[0]
        self.p(target).revealed = True

        self.log(
            f"Hòa phiếu! Vật tế thần {self.name(target)} chết thay."
        )

        yield from self.resolve_deaths({
            target: "Vật tế thần",
        })
        return

    # Hòa lượt đầu: biện hộ đủ 30 giây.
    yield from v4_defense(self, tied)

    # Tất cả người còn quyền vote được bỏ phiếu,
    # nhưng chỉ được chọn trong nhóm ứng viên hòa.
    counts = yield from self.collect_votes(
        tied,
        runoff=True,
    )

    if not counts:
        self.log("Không có phiếu hợp lệ. Hôm nay không treo ai.")
        return

    highest = max(counts.values())
    final_tied = [
        i for i, count in counts.items()
        if count == highest
    ]

    if len(final_tied) == 1:
        yield from self.lynch(final_tied[0])
    else:
        self.log(
            "Vẫn hòa sau lượt bỏ phiếu lại. Hôm nay không treo ai."
        )

def v5_make_summary(g, events, messages, alive_at_start):
    talk_counts = _SummaryCounter()
    accusations = _SummaryGroups(set)
    edges = set()

    claimants = set()
    reports = _SummaryGroups(set)

    def label(pid):
        return g.name(pid)

    def labels(ids):
        return ", ".join(label(i) for i in sorted(ids))

    # Đếm phát biểu từ lịch sử, không phụ thuộc việc lưu bộ nhớ chat.
    for item in messages:
        text = item["text"]
        match = _chat_re.match(r"#(\d+)\s", text)

        if match:
            pid = int(match.group(1))
            if pid in g.people:
                talk_counts[pid] += 1

    for event in events:
        speaker = event["speaker"]
        target = event["target"]
        result = event["result"]

        if event["claim"]:
            claimants.add(speaker)

        if target not in g.people:
            continue

        if result is not None:
            reports[(speaker, target)].add(bool(result))

        if (
            speaker != target
            and (event["accusation"] or result is True)
        ):
            accusations[target].add(speaker)
            edges.add((speaker, target))

    ranked = sorted(
        accusations.items(),
        key=lambda item: (-len(item[1]), item[0]),
    )

    lines = [
        f"TÓM TẮT TRANH LUẬN — NGÀY {g.day}",
        (
            f"{len(messages)} tin nhắn • "
            f"{len(talk_counts)} người phát biểu • "
            f"{len(alive_at_start)} người sống khi bắt đầu"
        ),
        "",
        "1. CÁC HƯỚNG NGHI NGỜ CHÍNH",
    ]

    if ranked:
        for target, speakers in ranked[:5]:
            lines.append(
                f"• {label(target)}: bị {len(speakers)} người tố "
                f"({labels(speakers)})."
            )
        lines.append(
            "  Đếm người tố khác nhau, không đếm lặp câu. "
            "Đây không phải số phiếu hay xác suất là Sói."
        )
    else:
        lines.append("• Chưa ghi nhận lời tố có mục tiêu rõ ràng.")

    lines += ["", "2. NHẬN VAI VÀ KẾT QUẢ SOI ĐƯỢC TUYÊN BỐ"]

    if claimants:
        lines.append(
            "• Nhận Tiên tri trong buổi này: " + labels(claimants) + "."
        )
    else:
        lines.append("• Không có lời nhận Tiên tri mới trong buổi này.")

    if reports:
        for (speaker, target), results in sorted(reports.items()):
            if len(results) > 1:
                result_text = "vừa nói Sói, vừa nói không phải Sói"
            else:
                result_text = (
                    "Sói" if True in results else "không phải Sói"
                )

            lines.append(
                f"• {label(speaker)} nói soi {label(target)} → {result_text}."
            )
    else:
        lines.append("• Không ghi nhận kết quả soi được công bố.")

    lines.append(
        "  Tất cả thông tin trên là lời trong chat, có thể là nói dối."
    )

    lines += ["", "3. MÂU THUẪN CẦN THEO DÕI"]

    contradictions = []

    # So với cả lời công bố của những ngày trước.
    historical_reports = _SummaryGroups(set)

    for speaker, target, result in g.reports:
        historical_reports[(speaker, target)].add(bool(result))

    for speaker, target in reports:
        if len(historical_reports[(speaker, target)]) > 1:
            contradictions.append(
                f"{label(speaker)} từng đưa hai kết quả khác nhau "
                f"về {label(target)}."
            )

    targets_this_morning = {target for _, target in reports}

    for target in sorted(targets_this_morning):
        positive = set()
        negative = set()

        for (speaker, reported_target), results in historical_reports.items():
            if reported_target != target:
                continue
            if True in results:
                positive.add(speaker)
            if False in results:
                negative.add(speaker)

        if positive and negative and len(positive | negative) > 1:
            contradictions.append(
                f"Về {label(target)}: {labels(positive)} nói Sói; "
                f"{labels(negative)} nói không phải Sói."
            )

    current_claimants = [
        i for i, role in g.claims.items()
        if role == "S" and g.p(i).alive
    ]

    if len(current_claimants) > 1:
        contradictions.append(
            "Có nhiều người còn sống cùng nhận Tiên tri: "
            + labels(current_claimants)
            + ". Bộ luật chỉ có tối đa một Tiên tri."
        )

    if contradictions:
        lines.extend("• " + text for text in contradictions)
    else:
        lines.append("• Chưa phát hiện mâu thuẫn rõ trong dữ liệu đã ghi nhận.")

    lines.append(
        "  Mâu thuẫn là điểm cần kiểm tra, không tự động chứng minh ai là Sói."
    )

    lines += ["", "4. TRANH CÃI VÀ MỨC THAM GIA"]

    mutual = sorted({
        tuple(sorted((a, b)))
        for a, b in edges
        if (b, a) in edges
    })

    if mutual:
        for a, b in mutual[:5]:
            lines.append(f"• Tố qua lại: {label(a)} ↔ {label(b)}.")
    else:
        lines.append("• Không ghi nhận cặp tố qua lại rõ ràng.")

    silent = [
        i for i in alive_at_start
        if talk_counts[i] == 0
    ]

    quiet = [
        i for i in alive_at_start
        if talk_counts[i] == 1
    ]

    if silent:
        lines.append("• Chưa phát biểu: " + labels(silent) + ".")

    if quiet:
        lines.append("• Chỉ phát biểu một lần: " + labels(quiet) + ".")

    if talk_counts:
        top = talk_counts.most_common(3)
        lines.append(
            "• Nói nhiều nhất: "
            + ", ".join(
                f"{label(i)} ({count} lượt)" for i, count in top
            )
            + "."
        )

    lines += ["", "5. BA ĐIỂM QUẢN TRÒ CẦN NHỚ"]

    reminders = []

    if ranked:
        target, speakers = ranked[0]
        reminders.append(
            f"Hướng tố nổi bật là {label(target)}, "
            f"với {len(speakers)} người tố khác nhau."
        )
    else:
        reminders.append(
            "Chưa hình thành hướng tố rõ ràng; phiếu có thể phân tán."
        )

    if contradictions:
        reminders.append(contradictions[0])
    elif reports:
        reminders.append(
            "Có kết quả soi được công bố; nhớ đối chiếu với vai lộ về sau."
        )
    else:
        reminders.append(
            "Chưa có kết quả soi công khai để đối chiếu."
        )

    reminders.append(
        "Sắp bỏ phiếu: so kết quả vote với hướng tố ở trên; "
        "đổi phiếu không tự nó là bằng chứng thuộc phe Sói."
    )

    for number, reminder in enumerate(reminders, 1):
        lines.append(f"{number}) {reminder}")

    lines += [
        "",
        "Phạm vi: chỉ tổng hợp phát biểu công khai của buổi sáng này.",
        "Không đọc vai bí mật để kết luận ai nói thật.",
    ]

    return "\n".join(lines)


def v5_discussion(self):
    chat3_init(self)

    history_start = len(self.history)
    alive_at_start = self.live()[:]

    # Dùng bộ thu riêng để không bỏ sót khi bộ nhớ Chat V3 cắt bớt.
    self._v5_captured_events = []

    try:
        yield from _V5_OLD_DISCUSSION(self)
        captured = list(self._v5_captured_events)
    finally:
        self._v5_captured_events = None

    # Không tạo bản tóm tắt dành cho quản trò trong chế độ nhập vai.
    if not self.gm:
        return

    messages = [
        item
        for item in self.history[history_start:]
        if item.get("chat") and item.get("scope") is None
    ]

    summary = v5_make_summary(
        self,
        captured,
        messages,
        alive_at_start,
    )

    v5_show_summary(self.app, summary)

    # Ghi vào lịch sử kín để nút Xuất nhật ký hiện có cũng lưu tóm tắt.
    self.log(summary, scope="gm")

def v6_init(g):
    if hasattr(g, "_v6_claimed_enchanted"):
        return

    # Chỉ lưu lời khai công khai, không phải danh sách thật.
    g._v6_claimed_enchanted = set()

    # Dùng để hạn chế lặp cảnh báo trong cùng một ngày.
    g._v6_spoken = set()


def v6_say(self, speaker, text, **evidence):
    v6_init(self)

    _V6_OLD_SAY(self, speaker, text, **evidence)

    # Cho người chơi thật khai báo bằng câu rõ ràng.
    # Chỉ ghi nhận lời khai về chính người đang nói.
    lower = " ".join(text.lower().split())

    declarations = (
        "tôi bị mê hoặc",
        "mình bị mê hoặc",
        "tôi đã bị mê hoặc",
        "mình đã bị mê hoặc",
        "tôi cũng bị mê hoặc",
        "mình cũng bị mê hoặc",
    )

    # Yêu cầu câu bắt đầu như vậy để hạn chế hiểu nhầm
    # các câu trích dẫn hoặc giả định.
    if any(lower.startswith(prefix) for prefix in declarations):
        self._v6_claimed_enchanted.add(speaker)


def v6_piper_assessment(g, actor):
    """
    Đánh giá từ thông tin mà actor được phép biết.

    confirmed:
        Người actor trực tiếp nhận mặt trong nhóm mê hoặc.
    claimed:
        Người công khai nói mình bị mê hoặc, có thể nói dối.

    Không đọc vai ẩn của bất kỳ mục tiêu nào.
    """
    v6_init(g)

    a = g.p(actor)
    live = set(g.live())

    # Người thổi sáo biết mình là ai, không dùng đánh giá này
    # để tự tố bản thân.
    if a.role == "P":
        return {
            "risk": 0.0,
            "confirmed": set(),
            "claimed": set(),
            "candidates": [],
            "scores": {},
        }

    confirmed = set(a.enchanted_known) & live

    if a.enchanted:
        confirmed.add(actor)

    claimed = g._v6_claimed_enchanted & live

    # Thông tin trực tiếp có trọng lượng đầy đủ.
    # Lời khai chưa kiểm chứng chỉ có trọng lượng một phần.
    effective_count = float(len(confirmed))

    for person in claimed - confirmed:
        distrust = max(0.0, g.suspicion(actor, person))
        credibility = max(0.15, 0.65 - distrust * 0.06)
        effective_count += credibility

    total = len(live)

    # Kích hoạt theo dõi từ khoảng một nửa bàn;
    # tăng mạnh khi gần đủ tất cả người ngoài Thổi sáo.
    if total <= 1 or effective_count == 0:
        risk = 0.0
    else:
        threshold = max(1.0, (total - 1) * 0.5)
        span = max(1.0, (total - 1) - threshold)
        risk = max(
            0.0,
            min(1.0, (effective_count - threshold) / span),
        )

    # Còn hai người chưa bị mê hoặc theo thông tin trực tiếp:
    # rất đáng báo động, nhưng vẫn chưa biết ai là Thổi sáo.
    if confirmed and total - len(confirmed) <= 2:
        risk = max(risk, 0.95)

    candidates = [
        person
        for person in sorted(live)
        if (
            person != actor
            and person not in confirmed
            and person != a.lover
        )
    ]

    scores = {}

    for person in candidates:
        target = g.p(person)

        # Chỉ loại khỏi diện nghi ngờ khi vai ĐÃ CÔNG KHAI.
        if target.revealed and target.role != "P":
            continue

        # Sói biết đồng đội không thể là Người thổi sáo.
        if a.wolf and person in a.pack:
            continue

        score = 1.0

        # Tự khai bị mê hoặc làm giảm nghi ngờ,
        # nhưng không được coi là bằng chứng tuyệt đối.
        if person in claimed:
            score -= 0.35

        # Tín hiệu hành vi chỉ đóng vai trò phụ.
        score += min(
            0.35,
            max(0.0, g.suspicion(actor, person)) * 0.04,
        )

        if target.revealed and target.role == "P":
            score += 5.0

        scores[person] = score

    return {
        "risk": risk,
        "confirmed": confirmed,
        "claimed": claimed,
        "candidates": list(scores),
        "scores": scores,
    }


def v6_bot_talk(self, actor, event_id=None):
    v6_init(self)

    a = self.p(actor)

    # Không chiếm lượt trả lời trực tiếp người chơi.
    if event_id is not None:
        return _V6_OLD_TALK(self, actor, event_id=event_id)

    key = (self.day, actor)

    # Chỉ một lượt chuyên đề phe thứ ba mỗi bot mỗi ngày.
    if key in self._v6_spoken:
        return _V6_OLD_TALK(self, actor)

    # Giữ ưu tiên công bố Sói của Tiên tri thật.
    has_wolf_result = any(
        result
        and self.p(person).alive
        and person != a.lover
        for person, result in a.seen.items()
    )

    seer_not_announced = (
        (self.day, actor, "seer")
        not in getattr(self, "_ai2_announcements", set())
    )

    if a.role == "S" and has_wolf_result and seer_not_announced:
        return _V6_OLD_TALK(self, actor)

    # Thổi sáo có thể giả làm nạn nhân.
    if a.role == "P":
        if (
            self._v6_claimed_enchanted
            and actor not in self._v6_claimed_enchanted
            and self.rng.random() < 0.4
        ):
            self._v6_spoken.add(key)

            self.say(
                actor,
                "Mình cũng bị mê hoặc. Đừng chỉ chăm chăm tìm Sói, "
                "nhưng cũng đừng lấy chuyện ai chưa khai "
                "để kết luận người đó là Thổi sáo.",
            )
            return

        return _V6_OLD_TALK(self, actor)

    assessment = v6_piper_assessment(self, actor)
    confirmed = assessment["confirmed"]

    # Nạn nhân công khai lời khai về bản thân.
    if a.enchanted and actor not in self._v6_claimed_enchanted:
        self._v6_spoken.add(key)

        count = len(confirmed)

        lines = [
            "Mình bị mê hoặc. "
            f"Mình đã nhận mặt {count} người còn sống trong nhóm bị mê hoặc, "
            "tính cả mình. Đừng để phe thứ ba thắng trong lúc làng cãi nhau.",

            "Tôi bị mê hoặc. "
            f"Nhóm mình biết có {count} người còn sống. "
            "Sói với Dân còn đang đấu thì Thổi sáo có thể ăn cả bàn đấy.",
        ]

        self.say(actor, self.rng.choice(lines))
        return

    # Không có bất kỳ thông tin nào thì không tự biết
    # Thổi sáo đã hành động đến đâu.
    if not confirmed and not assessment["claimed"]:
        return _V6_OLD_TALK(self, actor)

    if self.rng.random() > 0.55:
        return _V6_OLD_TALK(self, actor)

    self._v6_spoken.add(key)

    candidates = assessment["candidates"]
    risk = assessment["risk"]

    if not candidates:
        self.say(
            actor,
            "Thông tin mê hoặc đang thiếu hoặc có người khai gian. "
            "Chưa thể chốt Thổi sáo; đừng biến một lời tự khai thành bằng chứng.",
        )
        return

    target = max(
        candidates,
        key=lambda person: assessment["scores"][person]
        + self.rng.uniform(-0.2, 0.2),
    )

    target_name = self.name(target)

    if risk >= 0.65:
        lines = [
            f"Khoan, phe thứ ba đang nguy hiểm rồi. {target_name} "
            "vẫn nằm trong nhóm mình chưa xác nhận bị mê hoặc. "
            "Mình muốn kiểm tra hướng này, không phải kết luận chắc.",

            f"Vãi, cứ cãi Sói với Dân là Thổi sáo ăn cả bàn mất. "
            f"{target_name}, bạn đã bị mê hoặc chưa? "
            "Lời tự khai vẫn cần đối chiếu nhé.",
        ]
    else:
        lines = [
            f"Nhớ để ý Thổi sáo nữa. {target_name}, bạn đã bị mê hoặc chưa? "
            "Mình hỏi để ghép thông tin, không phải cứ chưa bị là có tội.",

            "Ai bị mê hoặc thì khai rõ đi. "
            "Nhưng người chưa bị hoặc chưa nói chưa chắc là Thổi sáo. "
            f"Mình muốn nghe thêm từ {target_name}.",
        ]

    # Không dùng accusation=True ở đây:
    # nghi Thổi sáo không đồng nghĩa với nghi Sói.
    self.say(
        actor,
        self.rng.choice(lines),
        target=target,
        accusation=False,
    )


def v6_bot_pick(
    self,
    actor,
    options,
    kind,
    maximum=1,
    meta=None,
):
    v6_init(self)

    ids = [key for key, _ in options]

    if not ids:
        return []

    a = self.p(actor)

    # Chỉ can thiệp vào lựa chọn loại người.
    # Không thay đổi soi, bảo vệ, mê hoặc hoặc phát thuốc.
    if a.role != "P" and kind in ("vote", "hunter", "wolf"):
        assessment = v6_piper_assessment(self, actor)
        risk = assessment["risk"]

        valid = [
            person for person in ids
            if person in assessment["scores"]
        ]

        if risk >= 0.45 and valid:
            # Lựa chọn cũ vẫn có giá trị:
            # không bỏ qua Sói đã được xác nhận chỉ vì lo phe ba.
            old_choice = _V6_OLD_PICK(
                self, actor, options, kind, maximum, meta
            )

            candidates = set(valid)
            candidates.update(old_choice)

            def combined_value(person):
                piper_score = assessment["scores"].get(person, 0.0)

                if kind == "wolf":
                    base = 2.5 if person in old_choice else 0.0
                else:
                    base = self.suspicion(actor, person)

                return (
                    base
                    + risk * 8.0 * piper_score
                    + self.rng.uniform(-0.2, 0.2)
                )

            return [max(candidates, key=combined_value)]

    return _V6_OLD_PICK(
        self, actor, options, kind, maximum, meta
    )

    is_witch = kind == "witch"
    had_heal = self.p(actor).heal if is_witch else False

    selected = yield from _V7_OLD_CHOOSE(

            "poisoned": action if isinstance(action, int) else None,

    return selected


def v7_witch_pick(self, actor, options, meta=None):
    a = self.p(actor)
    ids = [key for key, _ in options]
    victim = (meta or {}).get("victim")

    can_save = "save" in ids and victim is not None

    # --------------------------------------------------------
    # 1. TỰ CỨU / CỨU NGƯỜI YÊU
    # --------------------------------------------------------

    if can_save and victim == actor:
        return ["save"]

    if can_save and victim == a.lover:
        return ["save"]

    # --------------------------------------------------------
    # 2. CÂN NHẮC CỨU NGƯỜI KHÁC
    # --------------------------------------------------------

    save_value = -100.0

    if can_save:
        distrust = self.suspicion(actor, victim)

        # Giữ thuốc là lựa chọn mặc định, không tự động cứu.
        save_value = 0.5

        # Ít người: mất thêm một mạng có thể rất nguy hiểm.
        if len(self.live()) <= 6:
            save_value += 2.4
        elif len(self.live()) <= 9:
            save_value += 1.0

        # Người đã nhận Tiên tri và chưa bị nghi nhiều.
        # Đây là đánh giá lời nhận vai, không nhìn vai thật.
        if self.claims.get(victim) == "S":
            save_value += max(0.0, 3.0 - max(0.0, distrust))

            published = any(
                speaker == victim
                for speaker, target, result in self.reports
            )
            if published:
                save_value += 0.5

        # Uy tín tích lũy từ bằng chứng công khai.
        if distrust < -1.5:
            save_value += 1.1

        if distrust > 2.5:
            save_value -= 2.0

        if distrust > 5.0:
            save_value -= 2.0

        # Lời soi đáng tin cho rằng nạn nhân là Sói:
        # không bỏ qua chỉ vì nạn nhân bị cắn.
        for speaker, target, result in self.reports:
            if (
                target == victim
                and result
                and self.p(speaker).alive
                and self.suspicion(actor, speaker) < 1.0
            ):
                save_value -= 1.5
                break

    # --------------------------------------------------------
    # 3. CHỌN MỤC TIÊU ĐỘC
    # --------------------------------------------------------

    poison_targets = [
        i for i in ids
        if (
            isinstance(i, int)
            and i != actor
            and i != a.lover
            # Không phí độc vào người đã được báo sắp chết vì cắn.
            and i != victim
        )
    ]

    poison_target = None
    poison_value = -100.0

    # Nhận định phe ba từ thông tin được phép biết, nếu có V6.
    piper = v6_piper_assessment(self, actor)

    for target in poison_targets:
        value = self.suspicion(actor, target)

        credible_accusers = {
            speaker
            for speaker, reported_target, result in self.reports
            if (
                reported_target == target
                and result
                and self.p(speaker).alive
                and self.suspicion(actor, speaker) < 1.0
            )
        }

        if credible_accusers:
            value += 1.4

        # Nguy cơ Thổi sáo cũng có thể là lý do dùng độc.
        # Không dùng danh tính thật của mục tiêu.
        if piper["risk"] >= 0.7:
            value += (
                3.5
                * piper["risk"]
                * piper["scores"].get(target, 0.0)
            )

        value += self.rng.uniform(-0.15, 0.15)

        if value > poison_value:
            poison_value = value
            poison_target = target

    # Bàn càng ít người, càng khó chờ thêm một vòng.
    poison_threshold = 4.0 if len(self.live()) <= 6 else 4.8
    save_threshold = 3.4

    should_save = can_save and save_value >= save_threshold
    should_poison = (
        poison_target is not None
        and poison_value >= poison_threshold
    )

    # Mỗi đêm chỉ dùng một bình.
    if should_save and should_poison:
        # Chỉ bỏ phương án cứu khi hướng dùng độc mạnh hơn rõ rệt.
        if poison_value >= save_value + 2.5:
            return [poison_target]
        return ["save"]

    if should_save:
        return ["save"]

    if should_poison:
        return [poison_target]

    # Chưa đủ thông tin: giữ cả hai bình.
    return []

def v7_bot_talk(self, actor, event_id=None):

    a = self.p(actor)

    # Không chiếm lượt phản hồi trực tiếp người chơi.
    if event_id is not None:
        return _V7_OLD_TALK(self, actor, event_id=event_id)

    if a.role != "F":
        return _V7_OLD_TALK(self, actor)

    key = (self.day, actor)

    if key in self._v7_witch_spoken:
        return _V7_OLD_TALK(self, actor)

    records = self._v7_witch_memory.get(actor, [])

    if not records:
        return _V7_OLD_TALK(self, actor)

    last = records[-1]

    if last["day"] != self.day:
        return _V7_OLD_TALK(self, actor)

    victim = last["victim"]
    saved = last["saved"]
    poisoned = last["poisoned"]

    # Phù thủy chỉ biết các lời tố công khai nhắm vào mình.
    under_pressure = any(
        speaker != actor
        and target == actor
        and result
        and self.p(speaker).alive
        for speaker, target, result in self.reports

    # không tự khai thuốc còn lại trong mọi buổi sáng.
    disclose = (
        under_pressure

            and self.suspicion(actor, saved) > 1.5
        )
        or (len(self.live()) <= 6 and self.rng.random() < 0.6)
        or self.rng.random() < 0.22
    )

            f"cho {self.name(saved)}. "
            "Mình là Phù thủy. Ai đang tố người đó thì đối chiếu thêm, "
            "đừng tự động tin hoặc tự động bỏ qua.",
        ]

    elif last["could_see_victim"] and victim is not None:
        lines = [
            f"Mình là Phù thủy. Mình đã thấy {self.name(victim)} "
            "là nạn nhân cú cắn nhưng không dùng bình cứu. "
            "Giữ thuốc cũng là một lựa chọn, không phải cứ thấy người chết "
            "là phải tiêu bình ngay.",

            f"Mình nói rõ: đêm qua mình biết "
            f"{self.name(victim)} bị chọn cắn. "
            "Mình chưa đủ tin để dùng bình cứu. "
            "Đừng đánh đồng việc không cứu với việc cùng phe Sói.",
        ]

    elif poisoned is not None:
        lines = [
            f"Mình nhận Phù thủy. Mình đã dùng độc với "
            f"{self.name(poisoned)} dựa trên hướng nghi ngờ trước đó. "
            "Có thể mình sai, nhưng đây là hành động mình chịu trách nhiệm.",

            f"Đừng mặc định mọi người chết đêm đều do Sói. "
            f"Mình là Phù thủy và đã đầu độc {self.name(poisoned)}. "
            "Đối chiếu thông tin công khai tiếp đi.",
        ]

    # Không có thông tin hữu ích thì tiếp tục tranh luận bình thường.
    if not lines:
        return _V7_OLD_TALK(self, actor)

    self._v7_witch_spoken.add(key)

    # Không gắn claim=True: bộ phân tích hiện dùng claim
    # riêng cho lời nhận Tiên tri.
    self.say(
        actor,
        chat3_unique(self, actor, lines),
    )

def v8_start(self):
    global DAY_TIME, DEFENSE_TIME, VOTE_TIME

    try:
        discussion = int(self._v8_discussion_var.get().strip())
        defense = int(self._v8_defense_var.get().strip())
        vote = int(self._v8_vote_var.get().strip())
    except ValueError:
        messagebox.showerror(
            "Thời gian không hợp lệ",
            "Hãy nhập thời gian bằng số nguyên, ví dụ: 90, 45, 20.",
        )
        return

    if not 10 <= discussion <= 600:
        messagebox.showerror(
            "Thời gian thảo luận",
            "Thời gian thảo luận sáng phải từ 10 đến 600 giây.",
        )
        return

    if not 10 <= defense <= 600:
        messagebox.showerror(
            "Thời gian biện hộ",
            "Thời gian biện hộ hòa phiếu phải từ 10 đến 600 giây.",
        )
        return

    if not 5 <= vote <= 300:
        messagebox.showerror(
            "Thời gian vote",
            "Thời gian mỗi lượt vote phải từ 5 đến 300 giây.",
        )
        return

    self._v8_time_settings = {
        "discussion": discussion,
        "defense": defense,
        "vote": vote,
    }

    # Engine hiện tại sử dụng ba biến này khi bắt đầu từng pha.
    DAY_TIME = discussion
    DEFENSE_TIME = defense
    VOTE_TIME = vote

    _V8_OLD_START(self)


def v8_collect_votes(self, candidates, runoff=False):
    self.set_phase(
        "Bỏ phiếu lại" if runoff else "Bỏ phiếu",
        day=True,
    )

    duration = float(VOTE_TIME)
    started = self.clock
    self.deadline = started + duration

    self.log(
        f"Thời gian bỏ phiếu tối đa: {VOTE_TIME} giây."
    )

    voters = [
        i for i in self.live()
        if not self.p(i).no_vote
    ]

    votes = {}

    # Thời điểm vote của bot tỷ lệ theo thời lượng đã chọn.
    # Bot dự kiến vote trong khoảng 10%–85% tổng thời gian.
    scheduled = {
        i: started + self.rng.uniform(
            duration * 0.10,
            duration * 0.85,
        )
        for i in voters
        if i != self.human
    }

    human_pending = self.human in voters
    self.answer = None

    if human_pending:
        choices = [
            i for i in candidates
            if i != self.human
        ]

        self.app.show_action(
            f"Chọn người treo — tối đa {VOTE_TIME} giây",
            self.options(choices),
            0,
            1,
            True,
        )

    def cast(voter, target):
        votes[voter] = target

        if target is None:
            self.log(
                f"{self.name(voter)} bỏ phiếu trắng."
            )
        else:
            self.log(
                f"{self.name(voter)} bỏ phiếu cho "
                f"{self.name(target)}."
            )

    while self.clock < self.deadline:
        if human_pending and self.answer is not None:
            target = self.answer[0] if self.answer else None

            cast(self.human, target)

            human_pending = False
            self.answer = None
            self.app.clear_action()

        ready = [
            i for i, when in scheduled.items()
            if self.clock >= when
        ]

        # Công bố theo thứ tự thời điểm bot quyết định.
        ready.sort(key=lambda i: scheduled[i])

        for voter in ready:
            choices = [
                i for i in candidates
                if i != voter
            ]

            selected = self.bot_pick(
                voter,
                self.options(choices),
                "vote",
            )

            cast(
                voter,
                selected[0] if selected else None,
            )

            del scheduled[voter]

        if len(votes) == len(voters):
            break

        yield 0.1

    # Người chưa bỏ phiếu khi hết giờ được tính phiếu trắng.
    for voter in voters:
        if voter not in votes:
            cast(voter, None)

    self.app.clear_action()
    self.answer = None
    self.deadline = None

    self.last_votes = dict(votes)

    self.ballots.append({
        "day": self.day,
        "runoff": runoff,
        "votes": dict(votes),
        "time_limit": VOTE_TIME,
    })

    counts = Counter(
        target
        for target in votes.values()
        if target is not None
    )

    if counts:
        summary = ", ".join(
            f"{self.name(target)}: {count}"
            for target, count in counts.most_common()
        )
        self.log("Tổng phiếu: " + summary)
    else:
        self.log("Không có phiếu hợp lệ.")

    yield 0.8
    return counts


App.menu = v8_menu
App.start = v8_start
Game.collect_votes = v8_collect_votes

_V9_OLD_PICK = Game.bot_pick
_V9_OLD_TALK = Game.bot_talk


def v9_init(g):
    if hasattr(g, "_v9_profiles"):
        return

    g._v9_profiles = {}
    g._v9_scan_last = {}
    g._v9_reveal_plan = {}
    g._v9_daily_bias = {}

    for person in g.people:
        # Phong cách độc lập với vai:
        # không thể dựa vào tính cách để xác định phe.
        g._v9_profiles[person] = {
            "caution": g.rng.uniform(0.25, 0.9),
            "curiosity": g.rng.uniform(0.25, 0.9),
            "secrecy": g.rng.uniform(0.3, 0.9),
        }


def v9_weighted(g, choices, weights):
    return g.rng.choices(
        choices,
        weights=[max(0.01, weight) for weight in weights],
        k=1,
    )[0]


def v9_bot_pick(
    self,
    actor,
    options,
    kind,
    maximum=1,
    meta=None,
):
    v9_init(self)

    ids = [key for key, label in options]

    if not ids:
        return []

    a = self.p(actor)
    profile = self._v9_profiles[actor]

    # --------------------------------------------------------
    # TIÊN TRI: THĂM DÒ, KHÔNG SOI THEO ĐÁM ĐÔNG LIÊN TỤC
    # --------------------------------------------------------

    if kind == "seer":
        fresh = [i for i in ids if i not in a.seen]

        # Không bỏ phí lượt soi lại khi vẫn còn người chưa soi.
        choices = fresh or ids

        # Nếu đã soi hết, hạn chế lặp đúng mục tiêu đêm trước.
        if not fresh and len(choices) > 1:
            last = self._v9_scan_last.get(actor)
            choices = [i for i in choices if i != last] or choices

        # Đầu ván chủ yếu khám phá ngẫu nhiên.
        exploration = (
            0.85 if self.day == 1
            else 0.65 if self.day == 2
            else 0.40
        )

        exploration += (profile["curiosity"] - 0.5) * 0.15

        if self.rng.random() < exploration:
            target = self.rng.choice(choices)
        else:
            weights = []

            for target in choices:
                suspicion = max(
                    0.0,
                    min(5.0, self.suspicion(actor, target)),
                )

                # Điểm nghi ngờ chỉ tăng nhẹ xác suất,
                # không luôn chọn người bị nghi cao nhất.
                weight = 1.0 + suspicion * 0.22

                # Nhận trùng Tiên tri là lý do kiểm tra hợp lệ,
                # nhưng người giả vai cũng có thể chỉ đang đánh lạc hướng.
                if self.claims.get(target) == "S":
                    weight += 0.65

                weights.append(weight)

            target = v9_weighted(self, choices, weights)

        self._v9_scan_last[actor] = target
        return [target]

    # --------------------------------------------------------
    # SÓI: KHÔNG LUÔN CẮN NGƯỜI NHẬN TIÊN TRI
    # --------------------------------------------------------

    if kind == "wolf":
        # Giữ chiến thuật xử lý phe ba của V6.
        assessment = v6_piper_assessment(self, actor)

        if assessment["risk"] >= 0.65:
            return _V9_OLD_PICK(
                self, actor, options, kind, maximum, meta
            )

        choices = [i for i in ids if i != a.lover] or ids
        weights = []

        for target in choices:
            weight = 1.0

            if self.claims.get(target) == "S":
                weight += 1.5

                # Người nhận Tiên tri có thể đang được bảo vệ.
                # Đây là suy đoán, không đọc mục tiêu Bảo vệ thật.
                weight *= 1.0 - 0.4 * profile["caution"]

            # Mức hoạt động là dữ kiện công khai.
            weight += min(1.0, self.p(target).talks * 0.06)
            weights.append(weight)

        return [v9_weighted(self, choices, weights)]

    # --------------------------------------------------------
    # CÁC KỸ NĂNG KHÁC: GIỮ QUYẾT ĐỊNH CHUYÊN BIỆT
    # --------------------------------------------------------

    selected = _V9_OLD_PICK(
        self, actor, options, kind, maximum, meta
    )

    if kind != "vote" or not selected:
        return selected

    # Không làm nhiễu quyết định quan trọng về phe ba.
    assessment = v6_piper_assessment(self, actor)

    if assessment["risk"] >= 0.45:
        return selected

    # Không phá chiến thuật riêng của tình nhân khác phe.
    if a.lover is not None and self.mixed_lovers():
        return selected

    original = selected[0]
    original_score = self.suspicion(actor, original)

    # Thông tin rất rõ: giữ lựa chọn cũ.
    if original_score >= 7.0:
        return selected

    candidates = [
        i for i in ids
        if (
            i != actor
            and i != a.lover
            and not (a.wolf and i in a.pack)
        )
    ]

    # Chỉ dao động giữa các mục tiêu có mức nghi ngờ gần nhau.
    close = [
        i for i in candidates
        if abs(self.suspicion(actor, i) - original_score) <= 0.9
    ]

    if len(close) < 2:
        return selected

    def decision_value(target):
        key = (self.day, actor, target)

        # Độ lệch riêng được giữ suốt ngày:
        # không đổi ý ngẫu nhiên ở mỗi lần gọi hàm.
        if key not in self._v9_daily_bias:
            self._v9_daily_bias[key] = self.rng.uniform(-0.55, 0.55)

        return (
            self.suspicion(actor, target)
            + self._v9_daily_bias[key]
        )

    return [max(close, key=decision_value)]


def v9_bot_talk(self, actor, event_id=None):
    v9_init(self)

    a = self.p(actor)

    # Các vai khác tiếp tục dùng hội thoại và chiến thuật V3–V7.
    if a.role != "S":
        return _V9_OLD_TALK(self, actor, event_id=event_id)

    # Không bỏ rơi người thật khi họ gọi tên.
    if event_id is not None:
        return _V9_OLD_TALK(self, actor, event_id=event_id)

    # Đã nhận vai công khai thì không giả vờ chưa từng nhận.
    if self.claims.get(actor) == "S":
        return _V9_OLD_TALK(self, actor)

    profile = self._v9_profiles[actor]

    known_wolves = [
        target
        for target, result in a.seen.items()
        if result and self.p(target).alive and target != a.lover
    ]

    # Chỉ đọc lời tố công khai.
    accused_by_scan = any(
        target == actor
        and result
        and self.p(speaker).alive
        for speaker, target, result in self.reports
    )

    fake_claimants = [
        i for i, role in self.claims.items()
        if role == "S" and i != actor and self.p(i).alive
    ]

    # Bị dồn trực tiếp hoặc bị người giả vai tranh quyền dẫn làng:
    # tăng lý do công bố, nhưng không mặc định lần nào cũng làm vậy.
    urgent = accused_by_scan or bool(fake_claimants)

    plan_key = (
        self.day,
        actor,
        bool(known_wolves),
        urgent,
    )

    if plan_key not in self._v9_reveal_plan:
        secrecy = profile["secrecy"]

        if known_wolves:
            probability = 0.65 - secrecy * 0.25
        else:
            probability = 0.12 - secrecy * 0.08

        if urgent:
            probability += 0.4

        if len(self.live()) <= 6:
            probability += 0.2

        probability = max(0.03, min(0.95, probability))

        self._v9_reveal_plan[plan_key] = (
            self.rng.random() < probability
        )

    reveal = self._v9_reveal_plan[plan_key]

    known = [
        target
        for target in a.seen
        if self.p(target).alive and target != a.lover
    ]

    if reveal and known:
        target = self.rng.choice(known_wolves or known)
        result = a.seen[target]

        # Không thay kết quả soi, không bịa bằng chứng.
        text = (
            f"Mình quyết định nói vai: mình là Tiên tri. "
            f"Đã soi {self.name(target)} "
            + ("ra Sói. " if result else "không phải Sói. ")
            + "Mọi người ghi lại để đối chiếu, "
            "đừng chỉ tin vì mình nhận vai."
        )

        self.say(
            actor,
            text,
            target=target,
            result=result,
            claim=True,
        )
        return

    # Giữ kín vai: vẫn nghe và phản biện người khác,
    # nhưng không gọi nhánh Tiên tri cũ vốn dễ tự lộ.
    event = chat3_find_event(self, actor)

    # Tránh dùng lời đối đáp mặc định có thể nói
    # “chưa đủ chắc” về người mình đã soi ra Sói.
    if event is not None and not known_wolves:
        if chat3_reply(self, actor, event):
            return

    others = [i for i in self.live() if i != actor]

    if not others:
        return

    target = self.suspect(actor, others)
    name = self.name(target)

    lines = [
        f"{name}, nói rõ hướng vote của bạn đi. "
        "Mình muốn nghe lý do trước khi chốt.",

        f"Mình đang xem {name}. "
        "Chưa muốn kể hết hướng nghĩ ngay, cứ để người này giải thích trước.",

        f"Khoan theo đám đông. {name}, "
        "bạn thấy ai đang dẫn phiếu sai hướng nhất?",

        f"{name}, nếu người bạn định treo không phải Sói "
        "thì ngày mai bạn sẽ kiểm tra hướng nào?",
    ]

    self.say(
        actor,
        chat3_unique(self, actor, lines),
        target=target,
        accusation=False,
    )

def v10_init(g):
    if hasattr(g, "_v10_guard_memory"):
        return

    g._v10_guard_memory = {}
    g._v10_bear_memory = {}
    g._v10_spoken = set()


def v10_choose(
    self,
    actor,
    title,
    options,
    kind,
    minimum=1,
    maximum=1,
    optional=False,
    meta=None,
):
    v10_init(self)

    selected = yield from _V10_OLD_CHOOSE(
        self,
        actor,
        title,
        options,
        kind,
        minimum=minimum,
        maximum=maximum,
        optional=optional,

    )

    if kind == "guard" and selected:

def v10_night(self):
    v10_init(self)

    yield from _V10_OLD_NIGHT(self)

    if self.done:
        return

    # Chụp thông tin tại bình minh.
    # Không tính lại hàng xóm tùy tiện giữa buổi tranh luận.
    live = self.live()

    for actor in self.role_ids("B"):
        position = live.index(actor)

        neighbors = sorted({
            live[position - 1],
            live[(position + 1) % len(live)],
        } - {actor})

        # Đây là phép kiểm tra của KỸ NĂNG GẤU,
        # không phải AI được đọc vai tùy ý.
        growl = any(self.p(i).wolf for i in neighbors)

        record = {
            "day": self.day,
            "neighbors": neighbors,
            "growl": growl,
            "alive_snapshot": set(live),
        }

        self._v10_bear_memory.setdefault(actor, []).append(record)

        names = ", ".join(self.name(i) for i in neighbors)

        self.private(
            actor,
            f"Thông tin Gấu tại bình minh: hàng xóm là {names}. "
            + (
                "Gấu gầm: ít nhất một hàng xóm là Sói."
                if growl
                else "Gấu không gầm: hai hàng xóm không phải Sói "
                     "tại thời điểm này."
            )
            + " Kỹ năng không nhận diện Người thổi sáo.",
        )


def v10_current_bear(g, actor):
    v10_init(g)

    records = g._v10_bear_memory.get(actor, [])

    if not records:
        return None

    record = records[-1]

    if record["day"] != g.day:
        return None

    # Sau một chuỗi chết, hàng xóm/phe có thể đã thay đổi.
    # Không coi dữ liệu bình minh là kết quả mới sau biến động.
    if record["alive_snapshot"] != set(g.live()):
        return None

    return record


def v10_suspicion(self, actor, target):
    value = _V10_OLD_SUSPICION(self, actor, target)

    a = self.p(actor)

    if (
        a.role != "B"
        or target == actor
        or target == a.lover
        or self.p(target).revealed
    ):
        return value

    record = v10_current_bear(self, actor)

    if record is None or target not in record["neighbors"]:
        return value

    if not record["growl"]:
        # Chỉ giảm nghi ngờ là Sói.
        # Bộ xử lý phe ba vẫn đánh giá mục tiêu riêng.
        return min(value, -3.5)

    others = [
        i for i in record["neighbors"]
        if i != target
    ]

    # Chỉ khi hàng xóm còn lại đã công khai không phải Sói
    # mới suy ra chắc hơn. Không dùng kết quả soi âm tính,
    # vì Sói ngụy trang vẫn qua mặt Tiên tri.
    others_publicly_clear = all(
        self.p(i).revealed and not self.p(i).wolf
        for i in others
    )

    if others_publicly_clear:
        return max(value, 10.0)

    # Gấu gầm không có nghĩa cả hai đều là Sói.
    return value + 2.3


def v10_bot_pick(
    self,
    actor,
    options,
    kind,
    maximum=1,
    meta=None,
):
    if kind != "guard":
        return _V10_OLD_PICK(
            self, actor, options, kind, maximum, meta
        )

    v10_init(self)

    a = self.p(actor)
    choices = [key for key, label in options]

    if not choices:
        return []

    weights = []

    for target in choices:
        distrust = self.suspicion(actor, target)
        weight = 1.0

        if target == actor:
            weight += 1.8

        if target == a.lover:
            weight += 5.0

        if self.claims.get(target) == "S":
            weight += max(0.0, 3.5 - max(0.0, distrust))

        if distrust < -1.5:
            weight += 0.8

        if distrust > 4:
            weight *= 0.35

        # Không đọc vai Thổi sáo để chọn bảo vệ.
        # Bảo vệ chỉ ngăn cú cắn, không ngăn mê hoặc.
        weights.append(max(0.05, weight))

    # Danh sách options do engine cung cấp đã loại
    # mục tiêu được bảo vệ đêm trước.
    selected = self.rng.choices(
        choices,
        weights=weights,
        k=1,
    )[0]

    return [selected]


def v10_bot_talk(self, actor, event_id=None):
    v10_init(self)

    a = self.p(actor)

    if a.role not in ("G", "B"):
        return _V10_OLD_TALK(self, actor, event_id=event_id)

    # Giữ ưu tiên trả lời người chơi thật.
    if event_id is not None:
        return _V10_OLD_TALK(self, actor, event_id=event_id)

    key = (self.day, actor)

    # Chỉ một lượt nói chuyên về kỹ năng mỗi ngày.
    if key in self._v10_spoken:
        return _V10_OLD_TALK(self, actor)

    if a.role == "B":
        record = v10_current_bear(self, actor)

        if record is None:
            return _V10_OLD_TALK(self, actor)

        neighbors = record["neighbors"]
        names = ", ".join(self.name(i) for i in neighbors)

        # Gấu gầm là thông tin quan trọng, thường nên công bố.
        # Gấu không gầm thì không nhất thiết phải lộ vai ngay.
        speak_chance = 0.85 if record["growl"] else 0.35

        if self.rng.random() > speak_chance:
            return _V10_OLD_TALK(self, actor)

        self._v10_spoken.add(key)

        if record["growl"]:
            target = self.suspect(actor, neighbors)

            lines = [
                f"Mình nhận Người thuần gấu. Sáng nay gấu gầm, "
                f"hai hàng xóm của mình là {names}. "
                "Ít nhất một người là Sói, chứ không phải chắc cả hai. "
                "Hai bạn giải thích hướng vote của mình đi.",

                f"Khoan cãi lan man. Mình là Người thuần gấu, "
                f"gấu đã gầm với nhóm hàng xóm {names}. "
                "Đây là hướng cần kiểm tra trước. "
                "Nhưng đừng tự chọn bừa một người rồi bảo chắc chắn.",
            ]

            self.say(
                actor,
                chat3_unique(self, actor, lines),
                target=target,
                accusation=True,
            )
        else:
            lines = [
                f"Mình là Người thuần gấu. Sáng nay gấu không gầm, "
                f"hàng xóm của mình là {names}. "
                "Theo kỹ năng, họ không phải Sói lúc bình minh. "
                "Nhưng vẫn có thể là Người thổi sáo, đừng bỏ qua phe ba.",

                f"Mình có thông tin từ Gấu: {names} không làm gấu gầm "
                "ở bình minh. Mình nhận Người thuần gấu. "
                "Thông tin này chỉ loại hướng Sói tại thời điểm đó, "
                "không chứng minh họ cùng phe Dân.",
            ]

            self.say(
                actor,
                chat3_unique(self, actor, lines),
            )

        return

    # --------------------------------------------------------
    # BẢO VỆ: BIẾT HÀNH ĐỘNG CỦA MÌNH, KHÔNG BIẾT CÚ CẮN
    # --------------------------------------------------------

    records = self._v10_guard_memory.get(actor, [])

    if not records or records[-1]["day"] != self.day:
        return _V10_OLD_TALK(self, actor)

    record = records[-1]
    target = record["target"]

    assessment = v6_piper_assessment(self, actor)

    # Khi phe ba nguy hiểm, nhắc giới hạn kỹ năng,
    # không tự bịa ra kết quả phát hiện Thổi sáo.
    if assessment["risk"] >= 0.65:
        self._v10_spoken.add(key)

        self.say(
            actor,
            chat3_unique(self, actor, [
                "Đừng nhầm được bảo vệ với được miễn mê hoặc. "
                "Bảo vệ chỉ chặn cú cắn, Thổi sáo vẫn hành động được. "
                "Ai có thông tin nhóm mê hoặc thì khai rõ.",

                "Phe ba đang đáng lo. Người sống qua đêm "
                "không tự nhiên thành người đã được xác minh. "
                "Cần tách chuyện Sói cắn và chuyện bị mê hoặc ra.",
            ]),
        )
        return

    # Không tự tiết lộ mục tiêu bảo vệ mỗi sáng.
    if self.rng.random() > 0.3:
        return _V10_OLD_TALK(self, actor)

    self._v10_spoken.add(key)

    if self.p(target).alive:
        lines = [
            f"Mình nhận Bảo vệ, đêm qua đã bảo vệ {self.name(target)}. "
            "Người đó còn sống, nhưng mình không biết Sói có cắn họ không. "
            "Đừng lấy chuyện này làm bằng chứng họ là Dân.",

            f"Mình đã bảo vệ {self.name(target)} đêm qua. "
            "Mình chỉ biết lựa chọn của mình, không thấy mục tiêu Sói. "
            "Nếu có người bảo mình chắc chắn cứu được thì chưa đúng đâu.",
        ]
    else:
        lines = [
            f"Mình nhận Bảo vệ. Đêm qua mình bảo vệ "
            f"{self.name(target)}, nhưng người đó vẫn chết. "
            "Kỹ năng không chặn độc, đạn hay chết theo tình nhân. "
            "Mình chưa đủ thông tin để kết luận nguyên nhân.",

            f"Mình đã bảo vệ {self.name(target)} mà người đó vẫn chết. "
            "Đừng mặc định mọi cái chết đều do cú cắn. "
            "Ai có thông tin khác thì đối chiếu, đừng đoán thành sự thật.",
        ]

    self.say(
        actor,
        chat3_unique(self, actor, lines),
    )

from collections import Counter as _EndCounter
from collections import defaultdict as _EndGroups

_V11_OLD_CHOOSE = Game.choose
_V11_OLD_VOTES = Game.collect_votes
_V11_OLD_FINISH = Game.finish


def v11_init(g):
    if hasattr(g, "_v11_actions"):
        return

    g._v11_actions = []
    g._v11_votes = []


def v11_choose(
    self,
    actor,
    title,
    options,
    kind,
    minimum=1,
    maximum=1,
    optional=False,
    meta=None,
):
    v11_init(self)

    role_before = self.p(actor).role
    option_names = dict(options)

    selected = yield from _V11_OLD_CHOOSE(
        self,
        actor,
        title,
        options,
        kind,
        minimum=minimum,
        maximum=maximum,
        optional=optional,
        meta=meta,
    )

    self._v11_actions.append({
        "day": self.day,
        "actor": actor,
        "role": role_before,
        "kind": kind,
        "title": title,
        "selected": list(selected),
        "description": (
            ", ".join(
                str(option_names.get(item, item))
                for item in selected
            )
            or "Không dùng"
        ),
    })

    return selected


def v11_collect_votes(self, candidates, runoff=False):
    v11_init(self)

    # Chụp vai ngay lúc bỏ phiếu, tránh đánh giá sai
    # khi Đứa trẻ hoang dã chuyển phe về sau.
    roles = {
        i: self.p(i).role
        for i in self.people
    }

    teams = {
        i: self.team(i)
        for i in self.people
    }

    counts = yield from _V11_OLD_VOTES(
        self,
        candidates,
        runoff=runoff,
    )

    leaders = []

    if counts:
        highest = max(counts.values())
        leaders = [
            i for i, count in counts.items()
            if count == highest
        ]

    self._v11_votes.append({
        "day": self.day,
        "runoff": runoff,
        "votes": dict(self.last_votes),
        "roles": roles,
        "teams": teams,
        "leaders": leaders,
    })

    return counts


def v11_build_reports(g, result):
    winner, reason = result

    def name(i):
        return g.name(i)

    def names(ids):
        return ", ".join(name(i) for i in sorted(ids))

    # Chỉ phân tích lịch sử trước lúc ghi báo cáo.
    history = list(g.history)

    public = [
        item for item in history
        if item["scope"] is None
    ]

    chats = [
        item for item in public
        if item["chat"]
    ]

    speeches = _EndCounter()
    chat_records = []

    for item in chats:
        match = re.match(r"#(\d+)\s", item["text"])

        if not match:
            continue

        speaker = int(match.group(1))

        if speaker not in g.people:
            continue

        speeches[speaker] += 1
        chat_records.append((speaker, item))

    # --------------------------------------------------------
    # CHI TIẾT CUỐI VÁN
    # --------------------------------------------------------

    detail = [
        "HỒ SƠ CUỐI VÁN — ĐÃ CÔNG KHAI BÍ MẬT",
        f"Kết quả: {winner} — {reason}",
        f"Số ngày: {g.day}",
        f"Còn sống: {len(g.live())}/{len(g.people)}",
        "",
        "1. NHÂN VẬT",
    ]

    for p in g.people.values():
        text = (
            f"• {name(p.id)}: {ROLE[p.role]} | "
            f"Phe cuối ván: {g.team(p.id)} | "
            f"{'Sống' if p.alive else 'Đã chết'}"
        )

        if p.original != p.role:
            text += f" | Ban đầu: {ROLE[p.original]}"

        if p.lover is not None:
            text += f" | Người yêu: {name(p.lover)}"

        if p.enchanted:
            text += " | Đã bị mê hoặc"

        detail.append(text)

    detail += ["", "2. PHIẾU BẦU TỪNG LƯỢT"]

    for ballot in g._v11_votes:
        round_name = (
            "Bỏ phiếu lại"
            if ballot["runoff"]
            else "Bỏ phiếu đầu"
        )

        detail.append(
            f"\nNgày {ballot['day']} — {round_name}"
        )

        totals = _EndCounter(
            target
            for target in ballot["votes"].values()
            if target is not None
        )

        for voter, target in ballot["votes"].items():
            if target is None:
                detail.append(f"  {name(voter)} → Phiếu trắng")
            else:
                detail.append(
                    f"  {name(voter)} → {name(target)} "
                    f"[{ROLE[ballot['roles'][target]]} lúc vote]"
                )

        detail.append(
            "  Tổng: "
            + (
                ", ".join(
                    f"{name(i)}: {count}"
                    for i, count in totals.most_common()
                )
                or "Không có phiếu hợp lệ"
            )
        )

    if not g._v11_votes:
        detail.append("• Ván kết thúc trước khi có lượt bỏ phiếu.")

    detail += [
        "",
        "3. LỰA CHỌN KỸ NĂNG",
        "Đây là lựa chọn, không tự động có nghĩa hành động thành công.",
        "Đề xuất cắn của từng Sói không nhất thiết là mục tiêu đàn chốt.",
    ]

    for action in g._v11_actions:
        detail.append(
            f"• Ngày {action['day']} — {name(action['actor'])} "
            f"[{ROLE[action['role']]}]: "
            f"{action['title']} → {action['description']}"
        )

    if not g._v11_actions:
        detail.append("• Không có lựa chọn kỹ năng được ghi nhận.")

    detail += ["", "4. DIỄN BIẾN CÔNG KHAI"]

    for item in public:
        detail.append(
            f"[Ngày {item['day']} · {item['phase']}] "
            f"{item['text']}"
        )

    # --------------------------------------------------------
    # THỐNG KÊ — KHÔNG DÙNG VAI CUỐI VÁN ĐỂ CHẤM PHIẾU CŨ
    # --------------------------------------------------------

    valid_votes = _EndCounter()
    wolf_votes = _EndCounter()
    same_civilian_votes = _EndCounter()
    piper_votes = _EndCounter()
    civilian_wolf_votes = _EndCounter()
    helped_lynch_wolf = _EndCounter()

    # Lấy người thực sự được thông báo treo,
    # không coi chỉ đứng đầu phiếu là đã bị treo.
    lynched = set()

    for item in public:
        match = re.search(
            r"Quyết định treo:\s*#(\d+)",
            item["text"],
        )
        if match:
            lynched.add((item["day"], int(match.group(1))))

    for ballot in g._v11_votes:
        for voter, target in ballot["votes"].items():
            if target is None:
                continue

            valid_votes[voter] += 1

            target_role = ballot["roles"][target]
            voter_team = ballot["teams"][voter]
            target_team = ballot["teams"][target]

            if target_role in ("W", "D"):
                wolf_votes[voter] += 1

                if voter_team == "Dân":
                    civilian_wolf_votes[voter] += 1

                    if (
                        len(ballot["leaders"]) == 1
                        and ballot["leaders"][0] == target
                        and (ballot["day"], target) in lynched
                    ):
                        helped_lynch_wolf[voter] += 1

            if target_role == "P":
                piper_votes[voter] += 1

            if voter_team == "Dân" and target_team == "Dân":
                same_civilian_votes[voter] += 1

    review = [
        "ĐÁNH GIÁ TỔNG THỂ VÁN",
        f"Kết quả: {winner} — {reason}",
        (
            f"{len(g.people)} người • {g.day} ngày • "
            f"{len(chats)} tin chat • "
            f"{len(g._v11_votes)} lượt bỏ phiếu"
        ),
        "",
        "Đây là phân tích theo dữ liệu, không phải bảng xếp hạng kỹ năng tuyệt đối.",
        "Thông tin bí mật chỉ được dùng để đối chiếu SAU khi ván kết thúc.",
        "",
        "1. NHỮNG ĐÓNG GÓP CÓ THỂ KIỂM CHỨNG",
    ]

    if civilian_wolf_votes:
        best = max(civilian_wolf_votes.values())
        leaders = [
            i for i, count in civilian_wolf_votes.items()
            if count == best
        ]

        review.append(
            f"• Tìm đúng hướng Sói khi thuộc phe Dân: "
            f"{names(leaders)} — mỗi người {best} phiếu vào Sói."
        )
    else:
        review.append(
            "• Không ghi nhận phiếu của phe Dân nhắm đúng Sói."
        )

    if helped_lynch_wolf:
        for i, count in helped_lynch_wolf.most_common(3):
            review.append(
                f"• {name(i)} góp phiếu trong {count} lượt "
                "Sói thực sự bị treo."
            )

    if piper_votes:
        review.append(
            "• Từng vote đúng Người thổi sáo: "
            + names(piper_votes)
            + ". Đây là đối chiếu sau ván, chưa chứng minh "
              "người vote đã nhận diện đúng vai lúc đó."
        )

    review += [
        "",
        "2. MỨC THAM GIA VÀ NHỮNG CUỘC ĐỐI ĐẦU",
    ]

    for i, count in speeches.most_common(3):
        review.append(
            f"• {name(i)}: {count} lượt chat."
        )

    review.append(
        "  Nói nhiều không đồng nghĩa suy luận tốt; "
        "người chết sớm có ít cơ hội phát biểu hơn."
    )

    # Dùng lời công bố soi có cấu trúc, không suy đoán
    # mọi câu nhắc tên đều là lời tố.
    edges = {
        (speaker, target)
        for speaker, target, result in g.reports
        if result and speaker != target
    }

    mutual = sorted({
        tuple(sorted((a, b)))
        for a, b in edges
        if (b, a) in edges
    })

    if mutual:
        for a, b in mutual:
            review.append(
                f"• Đối đầu kết quả soi: {name(a)} và {name(b)} "
                "đều từng nói soi người kia ra Sói."
            )
    else:
        review.append(
            "• Không có cặp cùng công bố soi đối phương ra Sói."
        )

    fake_seers = [
        i for i, claimed_role in g.claims.items()
        if claimed_role == "S" and g.p(i).original != "S"
    ]

    if fake_seers:
        review.append(
            "• Nhận Tiên tri nhưng không có vai này: "
            + names(fake_seers)
            + "."
        )
        review.append(
            "  Chỉ xác nhận hành vi giả vai; chưa đủ dữ liệu "
            "để kết luận lời nói dối đã quyết định chiến thắng."
        )

    review += ["", "3. KHOẢNH KHẮC ĐÁNG CHÚ Ý"]

    moment_markers = (
        "không có ai chết",
        "Hòa phiếu!",
        "Vẫn hòa",
        "kích hoạt súng",
        "là Chàng ngốc!",
        "Gấu gầm",
        "Gấu không gầm",
    )

    moments = [
        item for item in public
        if not item["chat"]
        and any(marker in item["text"] for marker in moment_markers)
    ]

    # Lấy cả đầu, giữa, cuối ván nếu có quá nhiều sự kiện.
    if len(moments) > 10:
        indexes = {
            round(i * (len(moments) - 1) / 9)
            for i in range(10)
        }
        moments = [moments[i] for i in sorted(indexes)]

    if moments:
        for item in moments:
            review.append(
                f"• Ngày {item['day']}: {item['text']}"
            )
    else:
        review.append("• Không ghi nhận sự kiện đặc biệt trong nhóm thống kê.")

    review += [
        "",
        "4. GÓC CHAT VUI",
        "Trích dẫn thật, chọn theo từ khóa; không khẳng định ai cũng thấy hài.",
    ]

    funny_words = (
        "vãi", "ủa", "mất ngủ", "căng quá",
        "phiếu thương hại", "nói to hơn",
        "ăn cả bàn", "lên giàn",
    )

    funny = []
    used_speakers = set()
    used_quotes = set()

    for speaker, item in chat_records:
        text = item["text"]

        if (
            speaker not in used_speakers
            and text not in used_quotes
            and any(word in text.lower() for word in funny_words)
        ):
            funny.append(item)
            used_speakers.add(speaker)
            used_quotes.add(text)

    # Trải đều qua các đoạn ván thay vì chỉ lấy phần đầu.
    if len(funny) > 5:
        indexes = {
            round(i * (len(funny) - 1) / 4)
            for i in range(5)
        }
        funny = [funny[i] for i in sorted(indexes)]

    if funny:
        for item in funny:
            review.append(
                f"• Ngày {item['day']}: “{item['text']}”"
            )
    else:
        review.append(
            "• Không tìm thấy câu phù hợp; không tự bịa tình huống hài."
        )

    # --------------------------------------------------------
    # NHẬN XÉT RIÊNG CHO NGƯỜI NHẬP VAI
    # --------------------------------------------------------

    if g.human is not None:
        actor = g.human
        p = g.p(actor)

        if winner == "Hòa":
            outcome = "Ván hòa."
        elif g.team(actor) == winner:
            outcome = "Bạn thắng theo phe của mình."
        else:
            outcome = "Bạn không thắng ván này."

        review += [
            "",
            "5. HỒ SƠ CỦA BẠN",
            outcome,
            f"• Vai ban đầu: {ROLE[p.original]}.",
            f"• Vai cuối ván: {ROLE[p.role]}.",
            f"• Phe cuối ván: {g.team(actor)}.",
            f"• Phát biểu: {speeches[actor]} lượt.",
            f"• Phiếu có mục tiêu: {valid_votes[actor]} lượt.",
            f"• Trong đó nhắm Sói: {wolf_votes[actor]} lượt.",
            f"• Nhắm Người thổi sáo: {piper_votes[actor]} lượt.",
        ]

        if same_civilian_votes[actor]:
            review.append(
                f"• Khi thuộc phe Dân, bạn có "
                f"{same_civilian_votes[actor]} phiếu vào người cùng phe Dân."
            )
            review.append(
                "  Đây là đối chiếu sau ván, không có nghĩa "
                "bạn có đủ thông tin để tránh mọi phiếu nhầm."
            )

        own_actions = [
            action for action in g._v11_actions
            if action["actor"] == actor
        ]

        review.append(
            f"• Đã ghi nhận {len(own_actions)} lượt chọn kỹ năng "
            "(gồm cả lựa chọn không dùng)."
        )

        review.append(
            "• Gợi ý xem lại: mở tab Chi tiết cuối ván, "
            "so thông tin có trước lượt vote với vai thật sau khi lộ. "
            "Đừng đánh giá quyết định chỉ bằng kết quả cuối cùng."
        )

    review += [
        "",
        "GIỚI HẠN ĐÁNH GIÁ",
        "• Không phong MVP tổng thể chỉ từ số phiếu hoặc số câu chat.",
        "• Chưa chấm chính xác hiệu quả Bảo vệ/Phù thủy từ các tác động chồng nhau.",
        "• Không coi lời nhận vai là đúng chỉ vì được nói chắc chắn.",
        "• Vai bí mật và lựa chọn đêm chỉ được công khai trong báo cáo sau ván.",
    ]

    return "\n".join(detail), "\n".join(review)

    app.put(review_box, review)

    detail_box.yview_moveto(0)
    review_box.yview_moveto(0)

    # Tự mở báo cáo sau khi công bố kết quả.
    notebook.select(review_tab)


def v11_finish(self, result):
    v11_init(self)

    already_built = getattr(self, "_v11_report_built", False)

    _V11_OLD_FINISH(self, result)

    if already_built:
        return

    self._v11_report_built = True

    detail, review = v11_build_reports(self, result)

    self._v11_detail_report = detail
    self._v11_review_report = review

    v11_show_reports(self.app, detail, review)

    # Ghi công khai SAU khi kết thúc:
    # nhập vai cũng xuất được báo cáo bằng nút Xuất nhật ký.
    self.log(
        "[BÁO CÁO CHI TIẾT CUỐI VÁN]\n" + detail
    )
    self.log(
        "[ĐÁNH GIÁ CUỐI VÁN]\n" + review
    )

_V12_OLD_HUMAN_CHAT = Game.human_chat
_V12_OLD_SEND_CHAT = App.send_chat
_V12_OLD_REFRESH_INFO = App.refresh_info


def v12_human_chat(self, text):
    # Quản trò hoặc người còn sống: giữ nguyên luật trước đó.
    if self.human is None or self.p(self.human).alive:
        return _V12_OLD_HUMAN_CHAT(self, text)

    # Không gửi khi tạm dừng hoặc ván đã kết thúc.
    if self.paused or self.done:
        return False

    text = text.strip()[:600]

    if not text:
        return False

    last_sent = getattr(self, "_v12_last_ghost_chat", -10.0)

    if self.clock - last_sent < 0.8:
        return False

    self._v12_last_ghost_chat = self.clock

    # QUAN TRỌNG:
    # Không dùng self.say() vì say() gọi bộ nghe của bot,
    # cập nhật nghi ngờ, lời nhận vai và hàng chờ phản hồi.
    #
    # Chỉ hiển thị và ghi nhật ký.
    self.log(
        f"👻 HỒN MA · {self.name(self.human)}: {text}",
        chat=True,
    )

    return True


def v12_send_chat(self):
    g = self.game

    is_ghost = (
        g is not None
        and g.human is not None
        and not g.p(g.human).alive
    )

    if not is_ghost:
        return _V12_OLD_SEND_CHAT(self)

    # Đi đường riêng để không bị luật chỉ ứng viên
    # hòa phiếu được phát biểu của V4 chặn chat hồn ma.
    if g.done:
        self.chat_hint.configure(
            text="Ván đã kết thúc. Bạn có thể xem báo cáo hoặc xuất nhật ký.",
            foreground=MUTED,
        )
        return

    if g.paused:
        self.chat_hint.configure(
            text="Game đang tạm dừng. Tiếp tục ván để gửi chat hồn ma.",
            foreground="#ffcf87",
        )
        return

    text = self.chat_var.get()

    if not text.strip():
        return

    if g.human_chat(text):
        self.chat_var.set("")
        self.chat_hint.configure(
            text=(
                "👻 Đã gửi chat hồn ma. "
                "Bot sống không dùng lời này để suy luận."
            ),
            foreground="#cbb5ff",
        )
    else:
        self.chat_hint.configure(
            text="Bạn vừa gửi quá nhanh, chờ một chút rồi gửi tiếp nhé.",
            foreground="#ffcf87",
        )


def v12_refresh_info(self):
    _V12_OLD_REFRESH_INFO(self)

    g = self.game

    if (
        g is None
        or g.human is None
        or g.p(g.human).alive
    ):
        return

    # Chỉnh nhãn cũ “chỉ theo dõi” để khớp chức năng mới.
    text = self.private_label.cget("text")

    text = text.replace(
        "Đã chết — chỉ theo dõi",
        "Đã chết — được chat hồn ma",
    )

    self.private_label.configure(text=text)

    # Chỉ cập nhật lời nhắc khi vừa chuyển sang trạng thái chết.
    if getattr(self, "_v12_ghost_game", None) is not g:
        self._v12_ghost_game = g

        self.chat_hint.configure(
            text=(
                "👻 Bạn vẫn được chat. "
                "Lời hồn ma chỉ để bình luận, không ảnh hưởng quyết định bot."
            ),
            foreground="#cbb5ff",
        )


Game.human_chat = v12_human_chat
App.send_chat = v12_send_chat
App.refresh_info = v12_refresh_info
